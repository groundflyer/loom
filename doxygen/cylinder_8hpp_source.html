38template <std::floating_point T>
39constexpr Intersection<T>
40intersect(const Ray<T>& ray, const Cylinder<T>& prim) noexcept
41{
42 const auto& [o, d] = ray;
43 const auto& [p, r, h] = prim;
44
45 const auto [ox, oy, oz] = o - p;
46 const auto [dx, dy, dz] = d;
47
48 const T a = dx * dx + dz * dz;
49 const T b = 2 * (ox * dx + oz * dz);
50 const T c = ox * ox + oz * oz - r * r;
51
52 const auto [hit, t] = quadratic(a, b, c);
53
54 const T y = oy + dy * t;
55
56 return {hit && y >= 0 && y <= h, t, Vec2<T>{}};
57}

59/// @brief Return cylinder normal.
60template <std::floating_point T>
61constexpr Vec3<T>
62normal(Vec3<T> pos, const Cylinder<T>& prim) noexcept
63{
64 Vec3<T> p = prim.position;
65 p[1] = 0;
66 pos[1] = 0;
67 return normalize(pos - p);
68}

70template <std::floating_point T,
71 template <typename, std::size_t> typename SContainer>
72constexpr Vec3<T>
73sample(const Vec2<T, SContainer>& s, const Cylinder<T>& prim) noexcept
74{
75 const auto [p, r, h] = prim;
76 const auto [t1, t2] = s;
77
78 const T theta = static_cast<T>(2) * Pi<T> * t2;
79 return p + Vec3<T>{std::sin(theta), 0, std::cos(theta)} * r +
80 Vec3<T>{0, t2 * h, 0};
81}

84constexpr Vec3<T>
85centroid(const Cylinder<T>& prim) noexcept
86{
87 Vec3<T> p = prim.position;
