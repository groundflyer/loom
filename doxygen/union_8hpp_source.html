140template <typename... Ts>
141 requires(std::conjunction_v<std::is_default_constructible<Ts>...> &&
142 typelist::of_same_size<type_sequence<Ts...>>)
143class ProxyUnion
144{
145 using types = type_sequence<Ts...>;
146
147 static constexpr std::size_t data_size = typelist::max_sizeof<types>;
148 std::byte* p_data;
149
150 template <typename T>
151 static constexpr bool has = typelist::elem<T, types>;
152
153 public:
154 ProxyUnion() = delete;
155 constexpr ProxyUnion(const ProxyUnion& other) noexcept :
156 p_data(other.p_data)
157 {
158 }
159 constexpr ProxyUnion(ProxyUnion&& other) noexcept : p_data(other.p_data) {}
160 constexpr ProxyUnion(std::byte* data) noexcept : p_data(data) {}
161
162 constexpr ProxyUnion&
163 operator=(const ProxyUnion& rhs) noexcept
164 {
165 for(std::size_t i = 0; i < data_size; ++i) p_data[i] = rhs.p_data[i];
166 return *this;
167 }
168
169 template <typename T>
170 constexpr ProxyUnion&
171 operator=(const T& rhs) noexcept
172 requires(has<T>)
173 {
174 ref<T>() = rhs;
175 return *this;
176 }
177
178 template <typename T>
179 constexpr explicit
180 operator T() const noexcept
181 requires(has<T>)

183 return ref<T>();
184 }
185
186 template <typename T>
187 constexpr T&
188 ref() noexcept

190 return *reinterpret_cast<T*>(p_data);
191 }
192
193 template <typename T>
194 constexpr const T&
195 ref() const noexcept
196 {
197 return *reinterpret_cast<const T*>(p_data);
198 }

200 friend void
201 swap(ProxyUnion a, ProxyUnion b) noexcept
202 {
203 for(std::size_t i = 0; i < data_size; ++i)
204 std::swap(a.p_data[i], b.p_data[i]);
205 }

220/// @brief Unsafe union of vectors.
221///
222/// UnionVector holds a contigous memory segment and provides a typed view to
223/// it.
224///
225/// Only one type can be accessed at a time as
226/// the segment isn't aligned to common size.
227///
228/// When the integrated @p Types have the same size the operator[] is available.
229///
230/// The number of elements of each type in type_sequence @p Types is always
231/// equal and accesible via @p count().
232///
233/// @tparam Types type_sequence<Ts...>.
234template <typename Types, typename Allocator = std::allocator<std::byte>>
235 requires(typelist::all<std::is_default_constructible, Types>)
236class UnionVector
237{
238 template <typename T>
239 static constexpr bool has = typelist::elem<T, Types>;
240
241 static constexpr std::size_t m = typelist::max_sizeof<Types>;
242
243 std::size_t m_count{};
244 Allocator m_alloc = Allocator();
245 std::byte* p_data = nullptr;
246
247 _REINIT constexpr void
248 reinit() noexcept
249 {
250 m_count = 0;
251 p_data = nullptr;
252 }
253
254 constexpr void
255 allocate() noexcept
256 {
257 p_data = reinterpret_cast<std::byte*>(m_alloc.allocate(size_bytes()));
258 }
259
260 constexpr void
261 deallocate() noexcept
262 {
263 if(p_data) m_alloc.deallocate(p_data, size_bytes());
264 }
265
266 public:
267 /// @brief Construct an empty UnionVector.
268 constexpr UnionVector() noexcept = default;
269
270 /// @brief Construct an empty UnionVector with allocator @p alloc.
271 constexpr explicit UnionVector(const Allocator& alloc) noexcept :
272 m_alloc(alloc)
273 {
274 }
275
276 /// @brief Construct UnionVector with @p count of unitialized elements.
277 constexpr UnionVector(std::size_t count,
278 const Allocator& alloc = Allocator()) noexcept :
279 m_count(count),
280 m_alloc(alloc)
281 {
282 allocate();
283 }
284
285 /// @brief Construct UnionVector with @p count of elements initalized with
286 /// @p value of type @p T.
287 template <typename T>
288 constexpr UnionVector(std::size_t count,
289 const T& value,
290 const Allocator& alloc = Allocator()) noexcept
291 requires(has<T>)
292 : m_count(count), m_alloc(alloc)
293 {
294 allocate();
295 std::ranges::uninitialized_fill_n(reinterpret_cast<T*>(p_data),
296 m_count,
297 value);
298 }
299
300 constexpr UnionVector(const UnionVector& other) noexcept :
301 m_count(other.m_count), m_alloc(other.m_alloc)
302 {
303 allocate();
304 std::ranges::uninitialized_copy_n(other.p_data,
305 size(),
306 p_data,
307 p_data + size());
308 }
309
310 constexpr UnionVector(UnionVector&& other) noexcept :
311 m_count(other.m_count), m_alloc(other.m_alloc), p_data(other.p_data)
312 {
313 other.reinit();
314 }
315
316 constexpr UnionVector&
317 operator=(const UnionVector& rhs) noexcept
318 {
319 deallocate();
320
321 m_count = rhs.m_count;
322 m_alloc = rhs.m_alloc;
323
324 allocate();
325
326 std::ranges::uninitialized_copy_n(rhs.p_data,
327 size(),
328 p_data,
329 p_data + size());
330
331 return *this;
332 }
333
334 constexpr UnionVector&
335 operator=(UnionVector&& rhs) noexcept
336 {
337 deallocate();
338
339 m_count = rhs.m_count;
340 m_alloc = rhs.m_alloc;
341 p_data = rhs.p_data;
342
343 rhs.reinit();
344
345 return *this;
346 }
347
348 // operator[] depends on useless_t so it won't be instantiated if not asked
349 template <std::integral I>
350 constexpr const detail::useless_t<I, Types>
351 operator[](const I i) const noexcept
352 {
353 CHECK_INDEX(i, size());
354 return {p_data + (i * m)};
355 }
356
357 template <std::integral I>
358 constexpr detail::useless_t<I, Types>
359 operator[](const I i) noexcept
360 {
361 CHECK_INDEX(i, size());
362 return {p_data + (i * m)};
363 }
364
365 constexpr UnionVector&
366 resize(std::size_t count) noexcept
367 {
368 if(count != m_count)
369 {
370 deallocate();
371 m_count = count;
372 allocate();
373 }
374
375 return *this;
376 }
377
378 constexpr std::size_t
379 size() const noexcept
380 {
381 return m_count;
382 }
383
384 constexpr std::size_t
385 size_bytes() const noexcept
386 {
387 return m_count * m;
388 }
389
390 template <typename T>
391 constexpr std::span<T>
392 view() const noexcept
393 requires(has<T>)
394 {
395 return {const_cast<T*>(reinterpret_cast<const T*>(p_data)), m_count};
396 }
397
398 template <typename T>
399 constexpr std::span<T>
400 view() noexcept
401 requires(has<T>)
402 {
403 return {reinterpret_cast<T*>(p_data), m_count};
404 }
405
406 constexpr std::span<std::byte>
407 byteview() const noexcept
408 {
409 return {p_data, size()};
410 }
411
412 constexpr std::byte*
413 data() const noexcept
414 {
415 return p_data;
416 }
417
418 virtual ~UnionVector() { deallocate(); }
419};
