43template <std::floating_point T>
44auto
45gm_fz(RandomState& rs) noexcept
46{
47 constexpr T lbound = constants<T>::lbound;
48 constexpr T rbound = constants<T>::rbound;
49
50 return [g_a1 = [g_a = rs.make_gen<T>(1, rbound),
51 g_b = rs.make_gen<bool>(0.5)]() mutable noexcept
52 { return g_a() * (g_b() ? T(-1) : T(1)); },
53 g_f = rs.make_gen<T>(lbound, rbound)]() mutable noexcept
54 { return std::tuple{g_a1(), g_f(), g_f(), g_f()}; };
55}

57template <typename G>

59gen_input(unsigned num_runs, G& g) noexcept

67template <typename T, typename V>
68auto
69mk_val(const std::vector<T>& i, V v) noexcept
70{
71 auto l = [v](const auto& in) noexcept { return invopply(v, in); };
72 std::vector<decltype(l(i[0]))> ret;
73 ret.reserve(i.size());
74 std::ranges::transform(i, std::back_inserter(ret), l);
75 std::ranges::sort(ret);
76 return ret;
77}

87template <template <typename> typename _Context,
88 std::size_t threads_number,
89 bool affine_threads>
90void
91test_stream(std::string_view compiler_name,
92 Track& track,
93 const unsigned stream_size,
94 const unsigned num_runs,
95 const v_in_fac& in_fac,
96 const v_in_fib& in_fib,
97 const v_in_fz& in_fz,
98 const v_out_fac& out_val_fac,
99 const v_out_fib& out_val_fib,
100 const v_out_fz& out_val_fz) noexcept
101{
102 v_out_fac out_fac;
103 v_out_fib out_fib;
104 v_out_fz out_fz;
105
106 auto sort_outs = [&]() noexcept
107 {
108 std::ranges::sort(out_fac);
109 std::ranges::sort(out_fib);
110 std::ranges::sort(out_fz);
111 };
112 auto reset_outs = [&]() noexcept
113 {
114 out_fac.clear();
115 out_fib.clear();
116 out_fz.clear();
117
118 out_fac.reserve(num_runs);
119 out_fib.reserve(num_runs);
120 out_fz.reserve(num_runs);
121 };
122
123 auto oc = mp::make_output_collection(out_fac, out_fib, out_fz);
124
125 auto test_shop = [&](auto& shop) noexcept
126 {
127 auto ic = shop.bind_inputs(in_fac, in_fib, in_fz);
128 {
129 auto tp = shop.get_thread_pool();
130 if constexpr(affine_threads)
131 tp.affine(threads_number <=
132 std::thread::hardware_concurrency() / 2 ?
133 AFFINITY::PHYSICAL :
134 AFFINITY::FULL);
135
136 while(!mp::is_complete(ic)) shop.exchange(ic, oc);
137 shop.set();
138 }
139 shop.finalize(oc);
140 ASSERT(out_fac.size() == out_val_fac.size(), "sizes don't match");
141 ASSERT(out_fib.size() == out_val_fib.size(), "sizes don't match");
142 ASSERT(out_fz.size() == out_val_fz.size(), "sizes don't match");
143
144 sort_outs();
145
146 constexpr test::range_cmp_estimator<> rng_cmp{};
147
148 const double r_fac = rng_cmp(out_fac, out_val_fac);
149 const double r_fib = rng_cmp(out_fib, out_val_fib);
150 const double r_fz = rng_cmp(out_fz, out_val_fz);
151
152 return (r_fac + r_fib + r_fz) / 3.;
153 };
154
155 const bool really_affine =
156 affine_threads && threads_number <= std::thread::hardware_concurrency();
157 const std::string_view affine_s = really_affine ? "affine" : "";
158
159 track.check(NamedFuzzyTest(
160 [&]() noexcept
161 {
162 mp::Shop<programs, _Context, threads_number> shop(stream_size);
163 return test_shop(shop);
164 },
165 fmt::format("Static {} {} {} threads",
166 compiler_name,
167 threads_number,
168 affine_s),
169 0.01));
170
171 reset_outs();
172
173 track.check(NamedFuzzyTest(
174 [&]() noexcept
175 {
176 mp::Shop<programs, _Context, dynamic_size> shop(threads_number,
177 stream_size);
178 return test_shop(shop);
179 },
180 fmt::format("Dynamic {} {} {} threads",
181 compiler_name,
182 threads_number,
183 affine_s),
184 0.01));
185}
