57static bool
58val_stream_size(const std::size_t stream_size,
59 const std::size_t img_size) noexcept
60{
61 return stream_size * std::thread::hardware_concurrency() < img_size &&
62 stream_size > 64;
63}

70int
71main(int argc, char* argv[])
72{
73 auto parse_results = argparse::parse(
74 std::tuple{
75 argparse::option<'m', "method">(
76 "compute method, available options: hybrid, predicated, "
77 "scalar, reference",
78 "M",
79 HYBRID,
80 [](std::string_view sv) -> METHOD
81 {
82 if(sv == "hybrid") return HYBRID;
83 if(sv == "predicated") return PREDICATED;
84 if(sv == "scalar") return BSE;
85 if(sv == "reference") return REFERENCE;
86 throw UnknownMethod{sv};
87 }),
88 argparse::flag<'S', "no-mp">("disable multithreading", true),
89 argparse::option<'t', "threshold">("adaptive sampling threshold",
90 "T",
91 0.0001f,
92 read_number<float, true>),
93 argparse::option<'d', "depth">("maximum bounce depth",
94 "D",
95 20u,
96 read_number<unsigned, true>),
97 argparse::option<'b', "blur">("blur factor",
98 "B",
99 1.f,
100 read_number<float, true>),
101 argparse::option<'r', "res", 2>("image resolution",
102 {"W", "H"},
103 {400, 400},
104 read_number<unsigned, true>),
105 argparse::flag<'u', "unlock-res">("unlock image resolution", true),
106 argparse::option<'s', "stream-size">(
107 "stream size",
108 "N",
109 512ul,
110 read_number<std::size_t, true>),
111 argparse::option<'p', "exchange-period">(
112 "exchange period in milliseconds for multithreaded hybrid "
113 "method. Default: 1",
114 "T",
115 1,
116 read_number<unsigned, true>),
117 argparse::option<'f', "fps">("target fps. Default: 10",
118 "F",
119 10u,
120 read_number<unsigned, true>),
121 argparse::flag<'v', "verbose">("verbose logging")},
122 {argc, argv});
123
124 Logger logger(parse_results.template get_opt<'v'>() ? Logger::DEBUG :
125 Logger::INFO);
126
127 try
128 {
129 const auto [resx, resy] = parse_results.template get_opt<'r'>();
130 const bool lock_res = parse_results.template get_opt<'u'>();
131 const float threshold = parse_results.template get_opt<'t'>();
132 const unsigned max_depth = parse_results.template get_opt<'d'>();
133 const float blur = parse_results.template get_opt<'b'>();
134 std::size_t stream_size =
135 align_up(parse_results.template get_opt<'s'>(), 32ul);
136 const std::chrono::milliseconds ep{
137 parse_results.template get_opt<'p'>()};
138 const std::chrono::milliseconds tf{
139 1000u / parse_results.template get_opt<'f'>()};
140
141 if(threshold < 0)
142 throw std::runtime_error("Negative sampling threshold");
143
144 if(stream_size % simd::native_vector_extent_v<float> != 0)
145 throw std::runtime_error(
146 "Stream size is not aligned to SIMD lane size");
147
148 if(blur < 1.f || blur > 20.f)
149 throw std::runtime_error("Blur B have to be in range [1, 20]");
150
151 Image img(Vec2u{resx, resy});
152
153 if(!val_stream_size(stream_size, img.size()))
154 throw std::runtime_error("Stream size is too big");
155
156 logger.debug("exchange period: {}ms", ep.count());
157
158 EventState estate{img.size(),
159 stream_size,
160 threshold,
161 max_depth,
162 blur,
163 lock_res,
164 parse_results.template get_opt<'m'>(),
165 parse_results.template get_opt<'S'>()};
166
167 logger.debug("initializing glfw");
168 const auto& h = gl::System::init();
169
170 logger.debug("making window");
171 gl::Window window(resx,
172 resy,
173 resize_callback,
174 key_callback,
175 "Loom SmallPT Demo");
176
177 logger.debug("OpenGL version: {}; GLSL version: {}",
178 h.get_string(GL_VERSION),
179 h.get_string(GL_SHADING_LANGUAGE_VERSION));
180
181 logger.debug("making viewport");
182 gl::ImageViewport viewport{};
183
184 logger.debug("making osd");
185 gl::OSD osd(5.f, 5.f, osd_help, resx, resy);
186
187 resize_action = estate.get_resize_action(&osd);
188 key_action = estate.get_key_action();
189
190 viewport.load_img(img.data(), resx, resy);
191
192 ViewState vstate{h,
193 window,
194 viewport,
195 osd,
196 logger,
197 img,
198 threshold,
199 max_depth,
200 blur,
201 tf};
202
203 logger.debug("vstate ready");
204
205 logger.debug("image update timeframe: {}", vstate.tf.count());
206
207 while(!estate.req_quit)
208 {
209 vstate.update_stats(estate);
210 vstate.update_osd(img.size(), 0, 0);
211
212 if(estate.run_mp)
213 {
214 switch(estate.method)
215 {
216 case HYBRID: hybrid_mp(estate, vstate, ep); break;
217 case PREDICATED: predicated_mp(estate, vstate); break;
218 case BSE: bse_scanline_mp(estate, vstate); break;
219 case REFERENCE: reference_mp(estate, vstate); break;
220 }
221 }
222 else
223 {
224 switch(estate.method)
225 {
226 case HYBRID: hybrid(estate, vstate); break;
227 case PREDICATED: predicated(estate, vstate); break;
228 case BSE: bse_scanline(estate, vstate); break;
229 case REFERENCE: reference(estate, vstate); break;
230 }
231 }
232
233 logger.debug("final reload");
234 viewport.reload_img(img.data(), img.width(), img.height());
235 logger.debug("rendering done. show still image");
236 while(!estate.req_stop() && window.active())
237 {
238 viewport.draw();
239 osd.draw();
240 window.swap_buffers();
241 h.poll();
242 std::this_thread::sleep_for(100ms);
243 }
244
245 estate.resize(img);
246 logger.debug("new image size: {}x{}", img.width(), img.height());
247
248 vstate.threshold = estate.threshold;
249 vstate.max_depth = estate.max_depth;
250 vstate.blur = estate.blur;
251
252 logger.debug("stream size: {}", estate.stream_size);
253 logger.debug("threshold: {}", estate.threshold);
254 logger.debug("max depth: {}", estate.max_depth);
255 logger.debug("blur: {}", estate.blur);
256
257 estate.reset();
258 clear(img);
259 }
260
261 logger.debug("Goodbye");
262 }
263 catch(const std::runtime_error& ex)
264 {
265 logger.critical("{}", ex.what());
266 return 1;
267 }
268 catch(const std::length_error& ex)
269 {
270 logger.critical("{}", ex.what());
271 return 1;
272 }
273 catch(const gl::GLException& ex)
274 {
275 logger.critical("OpenGL error: {}", ex.what());
276 return 1;
277 }
278 catch(const ReadException& ex)
279 {
280 logger.critical("Failed to parse number {}", ex.sv);
281 return 1;
282 }
283 catch(const UnknownMethod& ex)
284 {
285 logger.critical("Unknown method: {}", ex.sv);
286 return 1;
287 }
288 return 0;
289}
