29template <unsigned def = 1000>
30std::pair<unsigned, unsigned>
31parse_test_args(int argc, char* argv[])
32{
33 const auto pr =
34 argparse::parse(std::tuple{argparse::option<'n', "numruns">(
35 "number of runs for parametric tests",
36 "N",
37 def,
38 read_number<unsigned, allow_throws>),
39 argparse::option<'s', "seed">(
40 "random seed",
41 "S",
42 0,
43 read_number<unsigned, allow_throws>)},
44 {argc, argv});
45
46#ifdef __EXCEPTIONS
47 return {pr.template get_opt<'n'>(), pr.template get_opt<'s'>()};
48#else
49 return {pr.template get_opt<'n'>().value(),
50 pr.template get_opt<'s'>().value()};
51#endif
52}
