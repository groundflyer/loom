101 std::array<std::uint32_t, 2> c{1, 11};
102
103 for(std::size_t i = 0; i < N; ++i)
104 {
105 vc0[i] = c[0];
106 vc1[i] = c[1];
107
108 c = bump_ctr(c);
109 }
110
111 ElapsedTimer<> timer;
112
113 for(std::size_t i = 0; i < size; i += N * 2)
114 std::tie(dest0, dest1, vc0, vc1) = f(vc0, vc1);
115
116 return timer.elapsed();
