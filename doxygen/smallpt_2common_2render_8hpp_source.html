58template <typename RenderPixel, typename ImgProj = std::identity>
59static bool
60scanline(EventState& estate,
61 ViewState& vstate,
62 ImgProj imgproj = std::identity{})
63{
64 ElapsedTimer<> img_timer, osd_timer, avg_timer;
65
66 std::size_t progress{}, tmp_rc{}, all_rc{};
67
68#ifdef DEMO_WRITE_PROGRESS
69 std::FILE* concatfile = std::fopen("scanline_concatfile.txt", "a+");
70
71 static constexpr std::string_view wpref = "scanline";
72
73 write_image(vstate.img, concatfile, wpref, 0ul, .0);
74#endif
75
76 return std::ranges::none_of(
77 imgproj(vstate.img),
78 [&](auto elem)
79 {
80 const auto [p, rc] = RenderPixel{vstate.img.res(),
81 estate.threshold,
82 estate.max_depth,
83 estate.blur}(elem);
84 progress += p;
85 tmp_rc += rc;
86
87 if(img_timer.has_expired(vstate.tf))
88 {
89 vstate.update(estate);
90#ifdef DEMO_WRITE_PROGRESS
91 write_image(
92 vstate.img,
93 concatfile,
94 wpref,
95 progress,
96 std::chrono::duration<double>(img_timer.restart())
97 .count());
98#else
99 img_timer.restart();
100#endif
101 }
102
103 if(osd_timer.has_expired(osd_timeframe))
104 {
105 vstate.update_stats(estate);
106 all_rc += tmp_rc;
107 const auto at = avg_timer.elapsed();
108 vstate.update_osd(
109 progress,
110 ray_traversal_rate(tmp_rc, osd_timer.restart()),
111 ray_traversal_rate(all_rc, at),
112 std::chrono::duration_cast<std::chrono::seconds>(
113 at));
114 tmp_rc = 0;
115 }
116
117 return progress == vstate.img.size() || estate.req_stop();
118 }) ||
119 [&]()
120 {
121 vstate.update(estate);
122 const auto at = avg_timer.elapsed();
123 vstate.update_osd(vstate.img.size(),
124 ray_traversal_rate(tmp_rc, osd_timer.elapsed()),
125 ray_traversal_rate(all_rc, at),
126 std::chrono::duration_cast<std::chrono::seconds>(at));
127
128#ifdef DEMO_WRITE_PROGRESS
129 write_image(vstate.img,
130 concatfile,
131 wpref,
132 progress,
133 std::chrono::duration<double>(img_timer.elapsed()).count());
134 std::fclose(concatfile);
135#endif
136
137 return estate.req_stop();
138 }();
139}

141template <typename RenderPixel, typename ImgProj = std::identity>
142static bool
143scanline_mp(EventState& estate,
144 ViewState& vstate,
145 ImgProj imgproj = std::identity{})
146{
147 struct Shared
148 {
149 std::atomic_size_t counter{};
150 std::atomic_size_t progress{};
151 std::atomic_size_t ray_count{};
152 std::atomic_flag flag{};
153 };
154
155 PADDED_VALUE(Shared, scc);
156
157 decltype(auto) r = imgproj(vstate.img);
158
159#ifdef DEMO_WRITE_PROGRESS
160 std::FILE* concatfile = std::fopen("scanline_mp_concatfile.txt", "a+");
161 ElapsedTimer<> img_timer;
162
163 static constexpr std::string_view wpref = "scanline_mp";
164
165 write_image(vstate.img, concatfile, wpref, 0ul, .0);
166#endif
167
168 thread_pool<dynamic_size> tp(
169 std::thread::hardware_concurrency(),
170 [&,
171 &c = scc.counter,
172 &s = scc.flag,
173 &progress = scc.progress,
174 &ray_count = scc.ray_count,
175 res = vstate.img.res(),
176 is = vstate.img.size()]()
177 {
178 constexpr std::size_t cs = 16;
179
180 while(!s.test(std::memory_order_relaxed))
181 {
182 const std::size_t b =
183 c.fetch_add(cs, std::memory_order_relaxed);
184
185 if(b >= is) return;
186
187 std::ranges::for_each(
188 std::views::counted(std::ranges::begin(r) + b, cs),
189 [res,
190 threshold = estate.threshold,
191 max_depth = estate.max_depth,
192 blur = estate.blur,
193 &progress,
194 &ray_count](auto view) noexcept
195 {
196 const auto [p, rc] =
197 RenderPixel{res, threshold, max_depth, blur}(view);
198 progress.fetch_add(p, std::memory_order_relaxed);
199 ray_count.fetch_add(rc, std::memory_order_relaxed);
200 });
201 }
202 });
203
204 tp.affine(AFFINITY::PHYSICAL);
205
206 ElapsedTimer<> osd_timer, avg_timer;
207
208 std::size_t all_rc{};
209
210 while(!estate.req_stop() &&
211 scc.progress.load(std::memory_order_relaxed) < vstate.img.size())
212 {
213 vstate.update(estate);
214
215#ifdef DEMO_WRITE_PROGRESS
216 write_image(vstate.img,
217 concatfile,
218 wpref,
219 scc.progress.load(std::memory_order_relaxed),
220 std::chrono::duration<double>(img_timer.restart()).count());
221#endif
222
223 if(osd_timer.has_expired(osd_timeframe))
224 {
225 const std::size_t rc =
226 scc.ray_count.exchange(0, std::memory_order_relaxed);
227 all_rc += rc;
228 vstate.update_stats(estate);
229 const auto at = avg_timer.elapsed();
230 vstate.update_osd(
231 scc.progress.load(std::memory_order_relaxed),
232 ray_traversal_rate(rc, osd_timer.restart()),
233 ray_traversal_rate(all_rc, at),
234 std::chrono::duration_cast<std::chrono::seconds>(at));
235 }
236
237 std::this_thread::sleep_for(vstate.tf);
238 }
239
240 if(estate.req_stop())
241 {
242 scc.flag.test_and_set(std::memory_order_relaxed);
243 return false;
244 }
245
246 {
247 const std::size_t rc =
248 scc.ray_count.exchange(0, std::memory_order_relaxed);
249 all_rc += rc;
250 vstate.update(estate);
251 const auto at = avg_timer.elapsed();
252 vstate.update_osd(vstate.img.size(),
253 ray_traversal_rate(rc, osd_timer.elapsed()),
254 ray_traversal_rate(all_rc, at),
255 std::chrono::duration_cast<std::chrono::seconds>(at));
256 }
257
258#ifdef DEMO_WRITE_PROGRESS
259 write_image(vstate.img,
260 concatfile,
261 wpref,
262 scc.progress.load(std::memory_order_relaxed),
263 std::chrono::duration<double>(img_timer.elapsed()).count());
264 std::fclose(concatfile);
265#endif
266
267 return true;
268}
