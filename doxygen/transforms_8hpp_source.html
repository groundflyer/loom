39/// @brief Convert a vector from homogenous coordinates.
40/// @param a homogenous column vector.
41template <std::floating_point T,
42 template <typename, std::size_t> typename Container>
43constexpr Vec3<T>
44dehomogenize(const Vec4<T, Container>& a) noexcept
45{
46 Vec3<T> ret{a};
47 const T w = get_w(a);
48 if(!(almost_equal(w, static_cast<T>(0), 5) ||
49 almost_equal(w, static_cast<T>(1), 5)))
50 ret /= w;
51 return ret;
52}

54/// @brief Apply transform to a vector.
55/// @param t transform to apply.
56/// @param v vector to transform.
57template <std::floating_point T,
58 template <typename, std::size_t> typename MatContainer,
59 template <typename, std::size_t> typename VecContainer>
60constexpr Vec3<T>
61apply_transform(const Mat4<T, MatContainer>& t,
62 const Vec3<T, VecContainer>& v) noexcept
63{
64 return dehomogenize(t * Vec4<T>(v));
65}

67/// @brief Apply transform to a point.
68/// @param t transform to apply.
69/// @param p point to transform.
70template <std::floating_point T,

73constexpr Vec3<T>
74apply_transform_p(const Mat4<T, MatContainer>& t,
75 const Vec3<T, Vec3Container>& p) noexcept

80/// @brief Apply transform to a normal.
81/// @param t transform to apply.
82/// @param n noraml to transform.
83template <std::floating_point T,
84 template <typename, std::size_t> typename MatContainer,
85 template <typename, std::size_t> typename NormalContainer>
86constexpr Vec3<T>
87apply_transform_n(const Mat4<T, MatContainer>& t,
88 const Vec3<T, NormalContainer>& n) noexcept
89{
90 return normalize(dehomogenize(transpose(inverse(t)) * (Vec4<T>(n))));

93/// @brief Create translation transform.
94/// @param delta translation delta.
95template <std::floating_point T,
96 template <typename, std::size_t> typename Container>
97constexpr Mat4<T>
98translate(const Vec3<T, Container>& delta) noexcept

115/// @brief Create clockwise rotation transform.
116/// @param angle angle defined in radians.
117/// @param axis axis to rotate object around.
118template <std::floating_point T,
119 template <typename, std::size_t> typename Container>
120constexpr Mat4<T>
121rotate(const T angle, const Vec3<T, Container>& axis) noexcept
122{
123 const auto& [x, y, z] = axis;
124 const Mat3<T> K{0, -z, y, z, 0, -x, -y, x, 0};
125 return Mat4<T>(m3id<T> + K * sin(angle) +
126 K * K * (static_cast<T>(1) - cos(angle)),
127 0,
128 0,
129 0,
130 1);
131}

133/// @brief Create Look-At transform.
134/// @param eye eye position.
135/// @param target target position.
136/// @param up vector (usually {0, 1, 0}).
137template <std::floating_point T,
138 template <typename, std::size_t> typename EyeContainer,
139 template <typename, std::size_t> typename CenContainer,
140 template <typename, std::size_t> typename UpContainer>
141constexpr Mat4<T>
142look_at(const Vec3<T, EyeContainer>& eye,
143 const Vec3<T, CenContainer>& target,
144 const Vec3<T, UpContainer>& up) noexcept
145{
146 const Vec3<T> f = normalize(eye - target);
147 const Vec3<T> r = normalize(cross(up, f));
148 const Vec3<T> u = normalize(cross(f, r));
149 return inverse(Mat4<T>(r, 0, u, 0, f, 0, 0, 0, 0, 1) * translate(-eye));
150}

152/// @brief Build orthonormal basis for given normal (z axis).
153/// @cite Duff2017Basis
154/// @return a pair of normals representing x and y axes.
155template <std::floating_point T,
156 template <typename, std::size_t> typename Container>
157constexpr std::pair<Vec3<T>, Vec3<T>>
158basis(const Vec3<T, Container>& n) noexcept
159{
160 const auto& [nx, ny, nz] = n;
161 const T sign = std::copysign(static_cast<T>(1), nz);
162 const T a = static_cast<T>(-1) / (sign + nz);
163 const T b = nx * ny * a;
164 return std::pair(make_normal(static_cast<T>(1) + sign * nx * nx * a,
165 sign * b,
166 -sign * nx),
167 make_normal(b, sign + ny * ny * a, -ny));
168}

170/// @brief Build orthonormal basis for given normal (z axis).
171/// @cite Duff2017Basis
172/// @return 3x3 rotation matrix.
173template <std::floating_point T,
174 template <typename, std::size_t> typename Container>
175constexpr Mat3<T>
176basis_matrix(const Vec3<T, Container>& z) noexcept
177{
178 const auto [x, y] = basis(z);
179 return transpose(Mat3<T>(x, y, z));
180}
