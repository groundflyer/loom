39template <Kernel K, simd::STITCH stitch, simd::SORT sort_method, bool async>
40struct StreamPair
41{
42 std::string
43 get_name() const noexcept
44 {
45 return fmt::format("SIMD stream {} {}{}",
46 stitch_name<stitch>::chars,
47 sort_name<sort_method>::chars,
48 async ? " async" : "");

51 using stream_type =

147template <template <typename> typename KT,
148 template <typename> typename GM,
149 typename V,
150 bool do_reset = false,
151 bool complex = false>
152struct StandardTestSuite
153{
154 template <typename T>
155 bool
156 operator()(Track& track, ParmTestMaker tm, RandomState& rs) const noexcept
157 {
158 using k_t = KT<T>;
159 constexpr auto ti = TypeInfo<T>::chars;
160 constexpr std::string_view nameview = k_t::nameview;
161 constexpr GM<T> gm{};
162 constexpr V v{};
163
164 const unsigned vector_size = tm.num_runs;
165
166 StandardState<k_t, complex> state(vector_size);
167
168 auto g = gm(rs);
169
170 auto get_tfn = [&]<typename Test>(Test&& t) noexcept
171 {
172 if constexpr(std::is_same_v<std::remove_cvref_t<Test>,
173 bve::function<k_t, bve::INTROSORT>> ||
174 std::is_same_v<std::remove_cvref_t<Test>,
175 bve::function<k_t, bve::COUNTING_SORT>>)
176 return tm.rng_with_scalar_ref(g,
177 state.bve_func_is,
178 state.bse_func);
179 else
180 return tm.rng_with_ref(g,
181 std::forward<Test>(t),
182 state.bve_func_is);
183 };
184
185 auto reset_bve = [&](auto&& t) noexcept
186 {
187 auto tfn = get_tfn(t);
188 if constexpr(do_reset || complex)
189 return [&state, tfn = std::move(tfn)]() noexcept
190 {
191 bool r = tfn();
192 state.bve_func_is.reset();
193 return r;
194 };
195 else
196 return tfn;
197 };
198
199 auto get_bve_tests = [&]() noexcept
200 {
201 auto one = NamedTest(
202 reset_bve(state.bve_func_is),
203 fmt::format("{}: BVE function introsort [{}]", nameview, ti));
204 if constexpr(complex)
205 return std::tuple{
206 std::move(one),
207 NamedTest(reset_bve(state.bve_func_cs),
208 fmt::format("{}: BVE function counting sort [{}]",
209
210 nameview,
211 ti)),
212 NamedTest(reset_bve(state.bve_stream_is_w),
213 fmt::format("{}: BVE stream introsort [{}]",
214
215 nameview,
216 ti)),
217 NamedTest(reset_bve(state.bve_stream_cs_w),
218 fmt::format("{}: BVE stream counting sort [{}]",
219
220 nameview,
221 ti))};
222 else
223 return one;
224 };
225
226 auto get_simd_tests = [&]() noexcept
227 {
228 auto n = NamedTest(
229 reset_bve(state.simd_func_N_is),
230 fmt::format("{}: SIMD function N-order introsort [{}]",
231
232 nameview,
233 ti));
234 auto m = NamedTest(
235 reset_bve(state.simd_func_m_is),
236 fmt::format("{}: SIMD function minikernel introsort [{}]",
237
238 nameview,
239 ti));
240
241 if constexpr(complex)
242 return std::tuple_cat(
243 std::tuple{std::move(n), std::move(m)},
244 loom::map_tuple(state.simd_streams,
245 [&](const auto& sp) noexcept
246 {
247 return NamedTest(
248 reset_bve(sp.wrapper),
249 fmt::format("{}: {} [{}]",
250 nameview,
251 sp.get_name(),
252 ti));
253 }));
254 else
255 return std::pair{n, m};
256 };
257
258 return track.dep_check(
259 NamedTest(tm.with_ref(g, state.bse_func, v),
260 fmt::format("{}: BSE [{}]", nameview, ti)),
261 get_bve_tests(),
262 get_simd_tests());
263 }
264};
