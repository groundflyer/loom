82 auto [track, tm, rs] = test_init<100000>(argc, argv);

84 vm::bse::function<cross_dot_test<float>> bse_cross_dot;
85 vm::bse::function<cross_test<float>> bse_cross;
86 vm::bse::function<mat_vec_dot_test<float>> bse_mat_vec_dot;
87 vm::bse::function<const_mat_vec<float>> bse_cmat;
88 vm::bse::function<vec3_norm_test<float>> bse_vec3_norm;
89
90 auto gf = rs.make_gen<float>(-1e6, 1e6);
91 auto gf3 = [&]() noexcept { return std::tuple{gf(), gf(), gf()}; };
92 const auto g2v3 = [&]() noexcept
93 { return std::tuple{gf(), gf(), gf(), gf(), gf(), gf()}; };
94 const auto vec3_ulp_cmp = [](const auto& t, const auto& r) noexcept
95 {
96 return almost_equal(std::get<0>(t), r[0], ulp) &&
97 almost_equal(std::get<1>(t), r[1], ulp) &&
98 almost_equal(std::get<2>(t), r[2], ulp);
99 };
100
101 track.check(NamedTest(tm.invariant(g2v3, bse_cross_dot, std::identity{}),
102 "cross dot test"));
103
104 track.check(NamedFuzzyTest(tm.fuzzy_with_ref(
105 g2v3,
106 bse_cross,
107 [](float x0,
108 float x1,
109 float x2,
110 float y0,
111 float y1,
112 float y2) noexcept
113 {
114 const auto [z0, z1, z2] =
115 cross(Vec3f{x0, x1, x2},
116 Vec3f{y0, y1, y2});
117 return std::tuple{z0, z1, z2};
118 },
119 test::uni_tuple_cmp{ulp_cmp{ulp}}),
120 "cross test",
121 0.01));
122
123 track.check(NamedFuzzyTest(tm.fuzzy_with_ref(
124 [&]() noexcept
125 {
126 std::array<float, 12> ret;
127 for(std::size_t i = 0; i < 12; ++i)
128 ret[i] = gf();
129 return ret;
130 },
131 bse_mat_vec_dot,
132 [](const auto& arr) noexcept
133 {
134 const Mat3f m{arr};
135 const Vec3f v{arr[9], arr[10], arr[11]};
136 return dot(m, v);
137 },
138 vec3_ulp_cmp),
139 "matrix-vector product",
140 0.01));
141
142 track.check(NamedTest(tm.test(gf3, bse_cmat, std::equal_to<>{}),
143 "identity matrix multiplication"));
144
145 track.check(NamedFuzzyTest(tm.fuzzy_with_ref(
146 gf3,
147 bse_vec3_norm,
148 [](float v0, float v1, float v2) noexcept
149 {
150 const auto [r0, r1, r2] =
151 normalize(Vec3f{v0, v1, v2});
152 return std::tuple{r0, r1, r2};
153 },
154 test::uni_tuple_cmp{ulp_cmp{ulp}}),
155 "vec3 normalize test",
156 0.01));
157
158 return track.ret_code();
159}
