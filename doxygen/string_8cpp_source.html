90template <std::integral Int>
91std::string
92format_int(Int num, BASE ts, unsigned width) noexcept
93{
94 // libfmt-9 doesn't support dynamic type specifier
95 switch(ts)
96 {
97 case BIN: return fmt::format("{:0{}b}", num, width);
98 case OCT: return fmt::format("{:0{}o}", num, width);
99 case DEC: return fmt::format("{:0{}d}", num, width);
100 case HEX: return fmt::format("{:0{}x}", num, width);
101 }
102}

119base_to_int(BASE base)

130template <std::integral Int>
131bool
132test_integral(Track& track, const ParmTestMaker& tm, RandomState& rs) noexcept
133{
134 auto gi = rs.make_gen<Int>(std::numeric_limits<Int>::lowest(),
135 std::numeric_limits<Int>::max());
136 auto gb = [i = rs.make_gen<unsigned>(0, 3)]() mutable noexcept
137 { return static_cast<BASE>(i()); };
138 auto gb2 = [i = rs.make_gen<unsigned>(0, 2)]() mutable noexcept
139 { return static_cast<BASE>(i()); }; // octal has no prefix, ignore it
140 auto gw = rs.make_gen<unsigned>(0, 32);
141
142 auto cmp = [](std::optional<Int> t, Int v) noexcept
143 { return t && t.value() == v; };
144
145 auto gh1 = [&]() noexcept
146 {
147 Int num = gi();
148 BASE base = gb();
149 return std::pair{
150 std::pair{format_int(num, base, gw()), base_to_int(base)},
151 num
152 };
153 };
154
155 auto gh2 = [&]() noexcept
156 {
157 Int num = gi();
158 return std::pair{format_int_prefix(num, gb2(), gw()), num};
159 };
160
161 auto test_s2i = NamedTest{tm.with_hint(gh1, s2i<Int>, cmp),
162 fmt::format("s2i {}", TypeInfo<Int>::chars)};
163 auto test_s2i_bp =
164 NamedTest{tm.with_hint(gh2, s2i_base_prefix<Int>, cmp),
165 fmt::format("s2i_base_prefix {}", TypeInfo<Int>::chars)};
166
167 return track.dep_check(test_s2i, test_s2i_bp);
168}
