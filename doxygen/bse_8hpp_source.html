70template <Kernel K,
71 template <typename, typename, typename> typename _Memory = Memory,
72 typename GetResetVal =

74class MemoryFacade
75{
76 using reg_names = K::reg_names;
77 using reg_types = K::reg_types;
78 using pc_var = K::pc;
79 using input_vars = K::input_vars;
80 using output_vars = K::output_vars;
81 using index_t = K::index_t;
82
83 static constexpr auto reset_val = GetResetVal{}();
84
85 _Memory<reg_types, reg_names, pc_var> memory;
86
87 void
88 reset() noexcept
89 {
90 // no-op when is_straightline
91 if constexpr(!K::is_straightline) memory.pc() = reset_val;
92 }
93
94 public:
95 template <typename... Args>
96 requires(typelist::length<input_vars> == sizeof...(Args))
97 constexpr void
98 set_inputs(Args&&... args) noexcept
99 {
100 typelist::for_each_arg(
101 input_vars{},
102 [&]<typename Var>(auto&& arg) constexpr noexcept
103 { memory.template get<Var>() = arg; },
104 std::forward<Args>(args)...);
105 reset();
106 }
107
108 constexpr auto
109 get_outputs() const noexcept
110 {
111 return typelist::maybe_collect<std::tuple>(
112 output_vars{},
113 [&mem = memory]<typename Var>() constexpr noexcept
114 { return mem.template get<Var>(); });
115 }
116
117 template <typename Block, typename... Args>
118 constexpr decltype(auto)
119 step(Args&&... args) noexcept
120 {
121 return Block{}(memory, std::forward<Args>(args)...);
122 }
123
124 template <typename Block, typename... Args>
125 constexpr decltype(auto)
126 step(Block block, Args&&... args) noexcept
127 {
128 return block(memory, std::forward<Args>(args)...);
129 }
130
131 constexpr auto
132 pc() const noexcept
133 {
134 // no-op when is_straightline
135 if constexpr(!K::is_straightline) return memory.pc();
136 }

142/// @brief Specialized dispatch implementation.
143/// @tparam K kernel
144/// @tparam pc current PC
145/// @tparam Compiler compiler for basic block
146template <Kernel K,
147 typename K::index_t pc,
148 template <typename> typename Compiler>
149 requires(!allow_throws)
150class threaded_dispatch
151{
152 using blocks = K::blocks;
153 using succs = K::successors;
154 using index_t = K::index_t;
155
156 static constexpr index_t halt_pos = K::halt_pos;
157
158 template <index_t I>
159 using f = threaded_dispatch<K, I, Compiler>;
160
161 using spec = sd_helper<K,
162 typelist::at<pc, blocks>,
163 typelist::at<pc, succs>,
164 f,
165 Compiler>;
166
167 public:
168 template <typename _Memory>
169 static void
170 eval(_Memory& mem) noexcept
171 {
172 if constexpr(pc != halt_pos)
173 {
174 using ccss = spec::ccss;
175
176 const index_t next_pc = spec::eval_kernel(mem);
177
178 if constexpr(spec::l == 1) MUSTTAIL
179 return typelist::head<ccss>::eval(mem);
180 else if constexpr(spec::l == 2)
181 {
182 if(next_pc) MUSTTAIL
183 return snd_t<ccss>::eval(mem);
184 else MUSTTAIL return fst_t<ccss>::eval(mem);
185 }
186 else if constexpr(spec::l == 3) switch(next_pc)
187 {
188 MK_CASE(0);
189 MK_CASE(1);
190 MK_CASE(2);
191 }
192 else if constexpr(spec::l == 4) switch(next_pc)
193 {
194 MK_CASE(0);
195 MK_CASE(1);
196 MK_CASE(2);
197 MK_CASE(3);
198 }
199 else if constexpr(spec::l == 5) switch(next_pc)
200 {
201 MK_CASE(0);
202 MK_CASE(1);
203 MK_CASE(2);
204 MK_CASE(3);
205 MK_CASE(4);
206 }
207 else if constexpr(spec::l == 6) switch(next_pc)
208 {
209 MK_CASE(0);
210 MK_CASE(1);
211 MK_CASE(2);
212 MK_CASE(3);
213 MK_CASE(4);
214 MK_CASE(5);
215 }
216 else if constexpr(spec::l == 7) switch(next_pc)
217 {
218 MK_CASE(0);
219 MK_CASE(1);
220 MK_CASE(2);
221 MK_CASE(3);
222 MK_CASE(4);
223 MK_CASE(5);
224 MK_CASE(6);
225 }
226 else if constexpr(spec::l == 8) switch(next_pc)
227 {
228 MK_CASE(0);
229 MK_CASE(1);
230 MK_CASE(2);
231 MK_CASE(3);
232 MK_CASE(4);
233 MK_CASE(5);
234 MK_CASE(6);
235 MK_CASE(7);
236 }
237 else if constexpr(spec::l == 9) switch(next_pc)
238 {
239 MK_CASE(0);
240 MK_CASE(1);
241 MK_CASE(2);

256/// @brief Basic CFG evaluator.
257/// @tparam use_tbd use threaded block dispatch.
258template <Kernel K, bool use_tbd = true>
259class basic_scalar_compiler
260{
261 using labelmap = K::labelmap;
262 using blocks = K::blocks;
263 using index_t = K::index_t;
264
265 static constexpr index_t halt_pos = K::halt_pos;
266
267 template <typename C>
268 using compile_block = standard_compiler<C, labelmap>;
269
270 using compiled_blocks = typelist::map<compile_block, blocks>;
271
272 using facade = MemoryFacade<K>;
273
274 struct single_block_evaluator : private facade
275 {
276 template <typename... Args>
277 constexpr auto
278 operator()(const Args&... args) noexcept
279 {
280 facade::set_inputs(args...);
281 facade::template step<typelist::head<compiled_blocks>>();
282 return facade::get_outputs();
283 }
284 };
285
286 struct simple_evaluator : private facade
287 {
288 template <typename... Args>
289 constexpr auto
290 operator()(const Args&... args) noexcept
291 {
292 facade::set_inputs(args...);
293
294 if constexpr(use_tbd)
295 facade::step(
296 [](auto& mem) noexcept
297 { threaded_dispatch<K, 0, compile_block>::eval(mem); });
298 else
299 while(facade::pc() != halt_pos)
300 typelist::visit_at(
301 facade::pc(),
302 [&]<typename Block>() constexpr noexcept
303 { facade::template step<Block>(); },
304 compiled_blocks{},
305 sink);
306
307 return facade::get_outputs();
308 }
309 };
310
311 public:
312 using function = std::
313 conditional_t<K::is_straightline, single_block_evaluator, simple_evaluator>;
314};
