50// find_bounds
51// saves to: l, r
52// l = -10
53// r = 10
54// rc = root_condition
55// rc = rc != 0
56// pjmp2 rc loop end
57// loop:
58// l *= 10
59// r *= 10
60// rc = root_condition
61// rc = rc != 0

148template <std::floating_point T>
149constexpr T
150val_eval(T a, T b, T c, T d, T x) noexcept
151{
152 T x2 = x * x;
153 T x3 = x2 * x;
154 return loom::fma(
155 a,
156 x3,
157 loom::fma(b, x2, loom::fma(c, x, d))); // a * x3 + b * x2 + c * x + d;
158}

160template <std::floating_point T>
161constexpr bool
162val_root_condition(T a, T b, T c, T d, T l, T r) noexcept
163{
164 return std::signbit(val_eval(a, b, c, d, l)) !=
165 std::signbit(val_eval(a, b, c, d, r));
166}
167
168template <std::floating_point T>
169std::tuple<T, T>
170val_find_bounds(T a, T b, T c, T d) noexcept
171{
172 T l = -10;
173 T r = 10;
174
175 while(!val_root_condition(a, b, c, d, l, r))
176 {
177 l *= 10;
178 r *= 10;
179 }
180
181 return {l, r};
182}
183
184template <std::floating_point T>
185T
186median(const T l, const T r) noexcept
187{
188 return l + (r - l) * static_cast<T>(0.5);
189}

191template <std::floating_point T>
192std::tuple<T, loom::uint_of_size_t<sizeof(T)>>
193val_cubic_fz(T a, T b, T c, T d) noexcept
194{
195 auto [l, r] = val_find_bounds(a, b, c, d);
196 auto s = median(l, r);
197 auto y = abs(val_eval(a, b, c, d, s));
198 loom::uint_of_size_t<sizeof(T)> i = 0;
199 while(y >= constants<T>::err_threshold)
200 {
201 const bool rc = val_root_condition(a, b, c, d, l, s);
202 // if (rc)
203 // r = s;
204 // else
205 // l = s;
206 r = rc ? s : r;
207 l = rc ? l : s;
208
209 s = median(l, r);
210 y = abs(val_eval(a, b, c, d, s));
211 ++i;
212 }
213
214 return {s, i};
215}
