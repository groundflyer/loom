33/// @brief Test a @p property of @p testing function.
34/// @param n Number of runs.
35/// @param threshold Ratio of runs allowed to fail.
36/// @param property_name Label to display.
37/// @param generator Callable that generates a unique value each time its
38/// called.
39/// @param testing Function that takes the output of @p generator and returns
40/// some value.
41/// @param property Function that takes the outputs of @p generator and @p
42/// testing and asserts them returning bool.
43/// @returns true if test failed otherwise false.
44///
45/// Simplified approach to property-based testing.
46///
47/// Unlike classic unit testing, in property-based testing
48/// we test functions many times against various values.
49/// Properties are invariant, so it makes testing more robust.
50///
51/// First, we need to find a property of the testing function,
52/// or we can just use inverse functions, e.g. minus and plus thus
53/// testing two functions at the time.
54///
55/// Then we define our @p generator.
56/// It outputs random values in a valid range
57/// acceptable by @p testing function.
58///
59/// @p testing is the function we'd like to test.
60///
61/// If @p testing accepts more then one value,
62/// @p generator should return either a tuple or a pair.
63/// @p test_property will decide automatically how to call @p testing.
64///
65/// @p property checks if the property is preserved.
66/// It accepts the output of @p testing as the first argument
67/// and the output of @p generator as the second argument.
68/// In case of fail it returns true, otherwise false.
69///
70/// The test is probablistic: if the portion of failed tests
71/// is less than @p threshold, test passed.
72template <typename Testing, typename Property, typename Generator>
73bool
74test_property(const std::size_t n,
75 const double threshold,
76 std::string_view property_name,
77 Generator&& generator,
78 Testing&& testing,
79 Property&& property)
80{
81 std::size_t sum = 0;
82 for(std::size_t i = 0; i < n; ++i)
83 {
84 const auto feed = generator();
85 const bool result = property(loom::invopply(testing, feed), feed);
86 sum += static_cast<size_t>(result);
87 }
88
89 const double error = static_cast<double>(sum) / static_cast<double>(n);
90
91 const bool ret = error > threshold;
92 static const constexpr char RED[] = "\033[31m";
93 static const constexpr char GREEN[] = "\033[32m";
94 static const constexpr char RESET[] = "\033[0m";
95 if(ret)
96 fmt::print("{}FAIL{} {:.3f}% {}\n",
97 RED,
98 RESET,
99 error * 100,
100 property_name);
101 else
102 fmt::print("{}OK{} {:.3f}% {}\n",
103 GREEN,
104 RESET,
105 error * 100,
106 property_name);
107
108 return ret > 0;
109}
