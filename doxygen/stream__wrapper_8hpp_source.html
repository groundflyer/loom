33template <typename S>
34class StreamWrapper
35{
36 S* const p_stream;
37
38 public:
39 StreamWrapper(S* stream) noexcept : p_stream(stream) {}
40
41 template <typename Input>
42 auto
43 operator()(const Input& input) const noexcept
44 {
45 // prepare result container
46 using program = S::program;
47 std::vector<loom::vm::output_type<program>> out;
48 out.reserve(input.size());
49 auto inserter = std::back_inserter(out);
50
51 auto step = [s = this->p_stream]() noexcept
52 {
53 (*s)();
54 s->sort();
55 };
56
57 // set initial data
58 auto i = p_stream->set_inputs(input);
59 step();
60
61 // feed all the input data
62 while(i != std::ranges::end(input))
63 {
64 auto exs = p_stream->exchange(
65 ranges::subrange(i, std::ranges::end(input)));
66 i += std::ranges::size(exs);
67 std::ranges::copy(exs, inserter);
68 step();
69 }
70
71 // compute the rest
72 while(!p_stream->done()) step();
73
74 std::ranges::copy(p_stream->get_output_zipped(), inserter);
75
76 ASSERT(out.size() == input.size(), "Output size doesn't match");
77
78 return out;
79 }
80};
