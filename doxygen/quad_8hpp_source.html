36/// @brief Compute ray-quadrilateral intersection.
37///
38/// Implements algorithm described in @cite LD2004AERIT
39template <std::floating_point R>
40constexpr Intersection<R>
41intersect(const Ray<R>& ray, const Quad<R>& prim) noexcept
42{
43 using V3 = Vec3<R>;
44
45 const auto& [o, d] = ray;
46 const auto& [v00, v01, v11, v10] = prim;
47
48 const V3 e01 = v10 - v00;
49 const V3 e03 = v01 - v00;
50 const V3 p = cross(d, e03);
51 const R D = dot(e01, p);
52
53 if(abs(D) < std::numeric_limits<R>::min()) return {};
54
55 const V3 T = o - v00;
56 const R alpha = dot(T, p) / D;
57
58 constexpr range<R> range01{0, 1};
59
60 if(!range01(alpha)) return {};
61
62 const V3 Q = cross(T, e01);
63 const R beta = dot(d, Q) / D;
64
65 if(!range01(beta)) return {};
66
67 if((alpha + beta) > 1)
68 {
69 const V3 e23 = v01 - v11;
70 const V3 e21 = v10 - v11;
71 const V3 p_ = cross(d, e21);
72 const R D_ = dot(e23, p_);
73
74 if(abs(D_) < std::numeric_limits<R>::min()) return {};
75
76 const V3 T_ = o - v11;
77 const R alpha_ = dot(T_, p_) / D_;
78
79 if(alpha_ < 0) return {};
80
81 const V3 Q_ = cross(T_, e23);
82 const R beta_ = dot(d, Q_) / D_;
83 if(beta_ < 0) return {};
84 }
85
86 const R t = dot(e03, Q) / D;
87
88 if(t < 0) return {};
89
90 const V3 e02 = v11 - v00;
91 const V3 N = cross(e01, e03);
92 const V3 aN = abs(N);
93 const auto& [Nx, Ny, Nz] = N;
94 const auto& [aNx, aNy, aNz] = aN;
95
96 R a11{}, b11{};
97 if((aNx >= aNy) && (aNx >= aNz))
98 {
99 a11 = (e02[1] * e03[2] - e02[2] * e03[1]) / Nx;
100 b11 = (e01[1] * e02[2] - e01[2] * e02[1]) / Nx;
101 }
102 else if((aNy >= aNx) && (aNy >= aNz))
103 {
104 a11 = (e02[2] * e03[0] - e02[0] * e03[2]) / Ny;
105 b11 = (e01[2] * e02[0] - e01[0] * e02[2]) / Ny;
106 }
107 else
108 {
109 a11 = (e02[0] * e03[1] - e02[1] * e03[0]) / Nz;
110 b11 = (e01[0] * e02[1] - e01[1] * e02[0]) / Nz;
111 }
112
113 R u{}, v{};
114 if(abs(a11 - 1) < std::numeric_limits<R>::min())
115 {
116 u = alpha;
117 if(abs(b11 - 1) < std::numeric_limits<R>::min())
118 v = beta;
119 else
120 v = beta / (u * (b11 - 1) + 1);
121 }
122 else if(abs(b11 - 1) < std::numeric_limits<R>::min())
123 {
124 v = beta;
125 u = alpha / (v * (a11 - 1) + 1);
126 }
127 else
128 {
129 const R A = -(b11 - 1);
130 const R B = alpha * (b11 - 1) - beta * (a11 - 1) - 1;
131 const R C = alpha;
132 const R Delta = B * B - 4 * A * C;
133 const R QQ = static_cast<R>(-0.5) * (B + std::copysign(sqrt(Delta), B));
134 u = QQ / A;
135
136 if(!range01(u)) u = C / QQ;
137
138 v = beta / (u * (b11 - 1) + 1);
139 }
140
141 return {true, t, Vec2<R>(u, v)};
142}

144/// @brief Compute normal of a quadrilateral.
145template <std::floating_point T,
146 template <typename, std::size_t> typename PosContainer>
147constexpr Vec3<T>
148normal(const Vec3<T, PosContainer>&, const Quad<T>& prim) noexcept
149{
150 const auto e01 = std::get<3>(prim) - std::get<0>(prim);
151 const auto e03 = std::get<1>(prim) - std::get<0>(prim);
152 return normalize(cross(e03, e01));
153}

155/// @brief Sample a point on a quadrilateral surface.
156template <std::floating_point T,
157 template <typename, size_t> typename SContainer>
158constexpr Vec3<T>
159sample(const Vec2<T, SContainer>& s, const Quad<T>& prim) noexcept
160{
161 const auto& [a, c1, b, c2] = prim;
162 const auto& [t1, t2] = s;
163 const auto& c = t1 > static_cast<T>(0.5) ? c1 : c2;
164 return Vec3<T>(
165 detail::triangle_sample(Vec2<T>(resample(t1), t2), a, b, c));
166}

168/// @brief Compute centroid of a quadrilateral.
169template <std::floating_point T>
170constexpr Vec3<T>
171centroid(const Quad<T>& prim) noexcept
172{
173 const auto& [a, b, c, d] = prim;
174 return a + b + c + d / static_cast<T>(4);
175}
