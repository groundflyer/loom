29/// @brief Axis-aligned bounding box.
30template <std::floating_point T>
31struct AABB
32{
33 Vec3<T> vmin;
34 Vec3<T> vmax;
35
36 constexpr AABB() noexcept = default;
37
38 template <template <typename, std::size_t> typename Container1,
39 template <typename, std::size_t> typename Container2>
40 constexpr AABB(const Vec3<T, Container1>& _vmin,
41 const Vec3<T, Container2>& _vmax) noexcept :
42 vmin(_vmin), vmax(_vmax)
43 {
44 }
45
46 constexpr const auto&
47 operator[](const bool i) const noexcept
48 {
49 return i ? vmax : vmin;
50 }
51
52 constexpr auto&
53 operator[](const bool i) noexcept
54 {
55 return i ? vmax : vmin;
56 }
57
58 /// @brief Select vertex components individually.
59 /// @param idxs Vector of indicies (0, 1) to select.
60 ///
61 /// Creates vector containing selected components
62 /// of @p vmin (0) and @p vmax (1).
63 ///
64 /// This used in intersect function.
65 template <template <typename, std::size_t> typename Container1>
66 constexpr Vec3<T>
67 operator[](const Vector<bool, 3, Container1>& idxs) const noexcept
68 {
69 Vec3<T> ret{};
70 for(std::size_t i = 0; i < 3; ++i) ret[i] = (*this)[bool{idxs[i]}][i];
71 return ret;
72 }
73};

81/// @brief Axis-aligned bounding box intersection function.
82///
83/// Implements algorithm described in @cite 10.1145/1198555.1198748
84template <std::floating_point T>
85constexpr Intersection<T>
86intersect(const Ray<T>& ray, const AABB<T>& prim) noexcept
87{
88 const auto& [o, d] = ray;
89 const Vec3<T> inv_d = Vec3<T>{1} / d;
90 const auto sign = inv_d < static_cast<T>(0);
91 const Vec3<T> vmin = inv_d * (prim[sign] - o);
92 const Vec3<T> vmax = inv_d * (prim[!sign] - o);
93 const auto& [xmin, ymin, zmin] = vmin;
94 const auto& [xmax, ymax, zmax] = vmax;
95 const T tmin1 = std::max(ymin, xmin);
96 const T tmax1 = std::min(ymax, xmax);
97 const T tmin2 = std::max(zmin, tmin1);
98 const bool intersected =
99 xmin <= ymax && ymin <= xmax && tmin1 <= zmax && zmin <= tmax1;
100 return {intersected && !std::signbit(tmin2), tmin2, Vec2<T>{}};
101}

103/// @brief Compute normal of AABB at @p pos.
104template <std::floating_point T>
105constexpr Vec3<T>
106normal(const Vec3<T>& pos, const AABB<T>& prim) noexcept
107{
108 const T nsign[2]{-1, 1};
109 Vec3<T> vd[2]{};
110 T md[2]{};
111 for(unsigned i = 0; i < 2; ++i)
112 {
113 const Vec3 ad = loom::abs(pos - prim[bool(i)]);
114 md[i] = loom::min(ad);
115 vd[i] = Vec3(ad == md[i]);
116 }
117 const auto pp = md[0] > md[1];
118 return normalize(vd[pp] * nsign[pp]);
119}

123template <std::floating_point T>
124constexpr std::pair<Vec3<T>, Vec3<T>>
125diag(const AABB<T>& prim, const unsigned shift, const bool side) noexcept
126{
127 const auto& [vmin, vmax] = prim;
128 const Vec3<T>& startpoint = side ? vmin : vmax;
129 const Vec3<T> diag = side ? vmax - vmin : vmin - vmax;
130 return std::pair{startpoint,
131 startpoint + diag * roll(Vec3<T>(1, 1, 0), shift)};
132}
133} // namespace detail

135/// @brief Uniformly sample position on AABB surface.
136template <std::floating_point T,
137 template <typename, std::size_t> typename SContainer>
138constexpr Vec3<T>
139sample(const Vec2<T, SContainer>& s, const AABB<T>& prim) noexcept
140{
141 const auto& [s1, s2] = s;
142 const T s3 = fmod(static_cast<T>(2) * (s1 + s2), static_cast<T>(1));
143 const unsigned shift = static_cast<unsigned>(
144 static_cast<T>(300) * fmod(s1 + s2 + s3, static_cast<T>(1)));
145 const auto [a, b] = detail::diag(prim, shift, s3 > static_cast<T>(0.5));
146 return loom::transform(
147 [](const auto&... args) noexcept { return loom::lerp(args...); },
148 a,
149 b,
150 roll(Vec3<T>(resample(s1), resample(s2), static_cast<T>(0)), shift));
151}

153/// @brief Compute AABB centroid.
154template <std::floating_point T>
155constexpr Vec3<T>
156centroid(const AABB<T>& prim) noexcept
157{
158 const auto& [vmin, vmax] = prim;
159 return (vmin + vmax) * static_cast<T>(0.5);
160}
