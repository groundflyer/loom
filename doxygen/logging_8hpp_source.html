30/// @brief Simple logger with classic verbosity schema, timestamps and colorful
31/// output.
32///
33/// Uses fmt to format messages.
34///
35/// Prints only into stdout and stderr.
36class Logger
37{
38 public:
39 /// @brief Indicates what kind of messages should be printed.
40 ///
41 /// The lower the level the more messages will be printed.
42 /// For instance, if level is WARNING, only warnings, errors
43 /// and critical errors will be printed.
44 enum Level
45 {
46 DEBUG = 10, /*!< all kind of messages */
47 INFO = 20, /*!< without debug messages */
48 WARNING = 30, /*!< warning and errors */
49 ERROR = 40, /*!< errors */
50 CRITICAL = 50 /*!< only critical errors */
51 };
52
53 /// @brief Empty class tag type that used to indicate Logger
54 /// if the output line should be rewritten in place.
55 struct Flush
56 {
57 };
58
59 /// @brief An instance of Flush.
60 static constexpr Flush flush{};

62 Logger() = delete;
63
64 /// @param lvl_ Minimum verbosity level.
65 explicit Logger(const Level lvl_) : lvl(lvl_) {}
66
67 /// @brief Print a debug message to stdout.
68 ///
69 /// The message is printed only if the verbosity level is less or equal than
70 /// Level::DEBUG.
71 template <typename... Args>
72 void
73 debug(Args&&... args) const
74 {
75 if(lvl <= Level::DEBUG)
76 message<'D'>(stdout, fmt::fg(fmt::terminal_color::white), args...);
77 }
78
79 /// @brief Print an info message to stdout.
80 ///
81 /// The message is printed only if the verbosity level is less or equal than
82 /// Level::INFO.
83 ///
84 /// The message prefix is highlighted with green.
85 template <typename... Args>
86 void
87 info(Args&&... args) const
88 {
89 if(lvl <= Level::INFO)
90 message<'I'>(stdout, fmt::fg(fmt::terminal_color::green), args...);
91 }
92
93 /// @brief Print a warning to stderr.

96 /// Level::WARNING.
97 ///
98 /// The message prefix is highlighted with yellow.
99 template <typename... Args>
100 void
101 warning(Args&&... args) const

103 if(lvl <= Level::WARNING)

106
107 /// @brief Report an error to stderr.
108 ///
109 /// The message is printed only if the verbosity level is less or equal than
110 /// Level::ERROR.
111 ///
112 /// The message prefix is highlighted with red.
113 template <typename... Args>
114 void
115 error(Args&&... args) const
116 {
117 if(lvl <= Level::ERROR)

119 }
120
121 /// @brief Report a critical error to stderr.
122 ///
123 /// The message is printed only if the verbosity level is less or equal than
124 /// Level::CRITICAL.
125 ///
126 /// The message prefix is highlighted with bold red.
127 template <typename... Args>
128 void
129 critical(Args&&... args) const
130 {
131 if(lvl <= Level::CRITICAL)
132 message<'C'>(stderr,
133 fmt::emphasis::bold |
134 fmt::fg(fmt::terminal_color::red),
135 args...);
136 }
137
138 private:
139 Level lvl;

141 template <char i>
142 static void
143 timestamp(std::FILE* out, const fmt::text_style& ts) noexcept
144 {
145 const std::time_t date = std::time(nullptr);
146 fmt::print(out, ts, "[{:%T}][{}] ", *std::localtime(&date), i);
147 }
148
149 template <char i, bool r, typename... Args>
150 static void
151 submessage(std::FILE* out,
152 const fmt::text_style& ts,
153 std::string_view fs,
154 Args&&... args)
155 {
156 if constexpr(r) std::putc('\r', out);
157 timestamp<i>(out, ts);
158 fmt::print(out, fmt::runtime(fs), args...);
159 if constexpr(r)
160 std::fflush(out);
161 else
162 std::putc('\n', out);
163 }
164
165 template <char i, typename... Args>
166 static void
167 message(std::FILE* out, const fmt::text_style& ts, Args&&... args)
168 {
169 submessage<i, false>(out, ts, args...);
170 }
171
172 template <char i, typename... Args>
173 static void
174 message(std::FILE* out, const fmt::text_style& ts, Flush, Args&&... args)
175 {
176 submessage<i, true>(out, ts, args...);
177 }
178};
