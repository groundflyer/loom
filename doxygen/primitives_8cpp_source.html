54 using R = float;

61 constexpr R val_range = 1000000;
62 std::uniform_real_distribution big_dist(-val_range, val_range);
63 std::uniform_real_distribution rad_dist(0.1f, val_range);
64 std::bernoulli_distribution bern_dist(0.5);
65 auto randr = static_cast<R (*)(std::default_random_engine&)>(
66 std::generate_canonical<R, 8>);
67 auto sgen = [&]() noexcept { return V2{generate<2>(randr, g)}; };
68 auto posgen = [&]() noexcept { return V3(generate<3>(big_dist, g)); };
69 auto normgen = [&]() noexcept
70 { return std::apply(make_normal, generate<3>(big_dist, g)); };

72 const auto sample_intersect_property =
73 [](const auto& sampled, const auto& prim) noexcept
74 {
75 const auto& [target, origin] = sampled;
76 const Rr tohit{origin, target - origin};
77 const Rr tomiss{origin, origin - target};
78 const Ir hit = intersect(tohit, prim);
79 const Ir miss = intersect(tomiss, prim);
80 return !hit || miss ||
81 !almost_equal(hit.t, loom::distance(target, origin), 10);
82 };
83
84 const auto bound_property =
85 [](const auto& testing, const auto& prim) noexcept
86 {
87 const auto& [bound, ray] = testing;
88 const Ir hit_prim = intersect(ray, prim);
89 const Ir hit_bound = intersect(ray, bound);
90 return !hit_bound || hit_prim.t < hit_bound.t;
91 };
92
93 const auto test_prim =
94 [&](auto&& name, auto&& gen, auto&& sampler, auto&& property) noexcept
95 { return test_property(num_tests, 0.05, name, gen, sampler, property); };
96
97 const auto intersect_test_prim =
98 [&](auto&& name, auto&& gen, auto&& sampler) noexcept
99 { return test_prim(name, gen, sampler, sample_intersect_property); };
100
101 const auto sample_prim = [&](const auto& prim) noexcept
102 { return pair(loom::sample(sgen(), prim), posgen()); };
103
104 const auto sample_selfoccluded_prim = [&](const auto& prim) noexcept
105 {
106 const V3 sampled_point = loom::sample(sgen(), prim);
107 const V3 offset = sampled_point - centroid(prim);
108 const auto sign = transform([](const R val) noexcept
109 { return std::copysign(1.f, val); },
110 offset);
111 const V3 origin =
112 sampled_point + Vec3(generate<3>(rad_dist, g)) * sign + offset;
113 return pair(sampled_point, Vec3(origin));
114 };
115
116 const auto bound_gen = [&](const auto& prim) noexcept
117 {
118 const AABB<R> bbox = bound(prim);
119 const V3 sp = loom::sample(sgen(), prim);
120 const auto tb = Vector(generate<3>(bern_dist, g));
121 const auto offset = transform([](const auto& a, const auto& b) noexcept
122 { return a ? b : -b; },
123 tb,
124 Vec3(generate<3>(rad_dist, g)));
125 const V3 origin(bbox[tb] + offset);
126 return pair{bbox, Ray(origin, sp - origin)};
127 };

129 const auto bound_test_prim = [&](auto&& name, auto&& gen) noexcept

132 const auto aabb_gen = [&]() noexcept

138 const auto cyl_gen = [&]() noexcept

142 const auto quad_gen = [&]() noexcept
143 {
144 const V3 v00{0, 0, 0};
145 const V3 v01{0, rad_dist(g), 0};
146 const V3 v10{rad_dist(g), 0, 0};
147 const V3 v11 = v00 + v01 + v10;
148 const auto t =
149 dot(loom::rotate(big_dist(g), normgen()), translate(posgen()));
150 return apply_transform(t, Quad<R>{v00, v01, v11, v10});
151 };

153 const auto aabb_gen_l = [&](const R max_rad, const V3&) noexcept
154 {
155 return AABB(-Vec3(generate<3>(randr, g)) * max_rad * 0.5f,
156 Vec3(generate<3>(randr, g)) * max_rad * 0.5f);

159 { return Sphere(V3(0), randr(g) * max_rad); };
160 const auto disk_gen_l = [&](const R, const V3& n) noexcept
161 { return Disk(V3(0), n, rad_dist(g)); };
162 const auto triangle_gen_l = [&](const R, const V3& n) noexcept
163 {
164 const V3 v0{-rad_dist(g), -rad_dist(g), 0};
165 const V3 v1{0, rad_dist(g), 0};
166 const V3 v2{rad_dist(g), -rad_dist(g), 0};
167 M4 t(basis_matrix(n));
168 t[3][3] = 1;
169 return apply_transform(t, Triangle<R>{v0, v1, v2});
170 };
171 const auto quad_gen_l = [&](const R, const V3& n) noexcept
172 {
173 const R x = rad_dist(g);
174 const R y = rad_dist(g);
175 const V3 v00{-x, -y, 0};
176 const V3 v01{0, y * 2, 0};
177 const V3 v10{x * 2, 0, 0};
178 const auto v11 = v00 + v01 + v10;
179 M4 t(basis_matrix(n));
180 t[3][3] = 1;
181 return apply_transform(t, Quad<R>{v00, v01, v11, v10});
182 };

196 triangle_gen,
197 sample_prim);
198 ret += intersect_test_prim("Quad: sample/trace", quad_gen, sample_prim);
199 ret += intersect_test_prim("Cylinder sample/trace",
200 cyl_gen,
201 sample_selfoccluded_prim);
202
203 ret += bound_test_prim("Sphere: bound", sphere_gen);
204 ret += bound_test_prim("Disk: bound", disk_gen);
205 ret += bound_test_prim("Triangle: bound", triangle_gen);
206 ret += bound_test_prim("Quad: bound", quad_gen);
