726/// @brief Implements partion that guaranteed to use only swaps.
727template <std::input_iterator I,
728 std::predicate<typename std::iterator_traits<I>::value_type> P>
729I
730partition(I first, I last, P p) noexcept
731{
732 using helper_type = sort_helper<I>;
733
734 first = std::find_if_not(first, last, p);
735 if(first == last) return first;
736
737 for(auto i = first + 1; i != last; ++i)
738 if(p(*i)) helper_type::iter_swap(i, first++);
739
740 return first;
741}

743/// @brief Implements insertion sort that guaranteed to use only swaps.
744template <
745 std::random_access_iterator I,
746 std::predicate<typename sort_helper<I>::type, typename sort_helper<I>::type>
747 Cmp = std::less<typename sort_helper<I>::type>>
748void
749insertion_sort(I first, I last, Cmp cmp = Cmp{}) noexcept
750{
751 using helper_type = sort_helper<I>;
752
753 constexpr helper_type helper{};
754
755 for(auto i = first + 1; i != last; ++i)
756 for(auto j = i; (j != first) && cmp(helper(*j), helper(*(j - 1))); --j)
757 helper_type::iter_swap(j, j - 1);
758}

764/// @brief Implements quicksort that guaranteed to use only swaps.
765template <
766 std::random_access_iterator I,
767 std::predicate<typename sort_helper<I>::type, typename sort_helper<I>::type>
768 Cmp = std::less<typename sort_helper<I>::type>>
769void
770quicksort(I first, I last, Cmp cmp = Cmp{}) noexcept
771{
772 using value_type = std::iterator_traits<I>::value_type;
773 using difference_type = std::iterator_traits<I>::difference_type;
774 using helper_type = sort_helper<I>;
775
776 constexpr helper_type helper{};
777
778 if(first == last) return;
779
780 const difference_type len = std::distance(first, last);
781
782 if(len <= insertion_sort_limit<I>) return insertion_sort(first, last, cmp);
783
784 const auto pivot = helper(*(first + len / 2));
785
786 I middle1 = bve::partition(first,
787 last,
788 [=](value_type elem)
789 { return cmp(helper(elem), pivot); });
790 I middle2 = bve::partition(middle1,
791 last,
792 [=](value_type elem)
793 { return !cmp(pivot, helper(elem)); });
794
795 quicksort(first, middle1);
796 quicksort(middle2, last);
797}

799template <std::random_access_iterator I>
800 requires(std::three_way_comparable<typename sort_helper<I>::type>)
801I
802median_of_3(I ia, I ib, I ic) noexcept
803{
804 using helper_type = sort_helper<I>;
805
806 constexpr helper_type helper{};
807
808 const auto a = helper(*ia);
809 const auto b = helper(*ib);
810 const auto c = helper(*ic);
811
812 const bool alb = a < b;
813 const bool blc = b < c;
814 if(alb && blc) return ib;
815
816 const bool alc = a < c;
817 const bool cleb = c <= b;
818 if(alc && cleb) return ic;
819
820 const bool blea = b <= a;
821 if(alc && blea) return ia;
822
823 const bool clea = c <= a;
824 if(blc && clea) return ic;
825
826 if(clea && alb) return ia;
827
828 // if(cleb && blea)
829 return ib;
830}

832template <
833 std::random_access_iterator I,
834 std::predicate<typename sort_helper<I>::type, typename sort_helper<I>::type>
835 Cmp = std::less<typename sort_helper<I>::type>>
836void
837introsort_impl(I first, I last, const unsigned depth, Cmp cmp = Cmp{}) noexcept
838{
839 using value_type = std::iterator_traits<I>::value_type;
840 using difference_type = std::iterator_traits<I>::difference_type;
841 using helper_type = sort_helper<I>;
842
843 constexpr helper_type helper{};
844
845 if(first == last) return;
846
847 const difference_type len = last - first;
848
849 if(len <= insertion_sort_limit<I>) return insertion_sort(first, last, cmp);
850
851 if(depth == 0u) return quicksort(first, last, cmp);
852
853 const auto pivot = helper(*median_of_3(first, first + len / 2, last - 1));
854
855 I m = bve::partition(first,
856 last,
857 [=](value_type val) noexcept
858 { return cmp(helper(val), pivot); });
859
860 introsort_impl(first, m, depth - 1);
861 introsort_impl(m, last, depth - 1);
862}

888/// @brief Counting sort implementation.
889template <Kernel K, typename _Memory>
890fsize_t
891counting_sort(_Memory& in, _Memory& out) noexcept
892{
893 ASSERT(in.size() == out.size(), "Sizes must match");
894
895 using index_t = K::index_t;
896
897 constexpr fsize_t num_bins = K::num_blocks + 1;
898 constexpr index_t halt_pos = static_cast<index_t>(K::num_blocks);
899
900 index_t bins[num_bins] = {};
901
902 // histogram
903 for(index_t pc: in.pc()) ++bins[fit_in_bins(pc, halt_pos)];
904
905 const fsize_t ret = bins[num_bins - 1];
906
907 // prefix sum
908 for(index_t i = 1; i < num_bins; ++i) bins[i] += bins[i - 1];
909
910 // fill out
911 for(int i = in.size() - 1; i >= 0; --i)
912 out[--bins[fit_in_bins(in.pc()[i], halt_pos)]] = in[i];
913
914 return ret;
915}
