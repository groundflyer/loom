47/// @brief Make array of registers.
48/// @tparam P register identifer prefix.
49/// @tparam N array size.
50///
51/// Creates type_sequence of size @p N with registers identifiers.
52template <CharSequence P, std::size_t N>
53class make_reg_array
54{
55 template <std::uint8_t I>
56 using f = append_idx<P, I>;

58 public:
59 using type = intlist::map_t<f, std::make_index_sequence<N>>;

106/// @brief Generate macro of binary map with two vectors.
107/// @tparam Op operator. See available operators in vm/operators.hpp
108/// @tparam T variables type.
109/// @tparam Dst vector of destination registers.
110/// @tparam A vector or scalar for the first operand.
111/// @tparam B vector or scalar for the second operand.
112///
113/// At least one of @p A and @p B should be vector.
114template <typename Op, typename T, typelist::List Dst, typename A, typename B>
115 requires((typelist::List<A> || typelist::List<B>) &&
116 !(CharSequence<A> && CharSequence<B>))
117class make_bin_map
118{
119 template <std::uint8_t I>
120 using f = Compute<Op,
121 Variable<T, typelist::at<I, Dst>>,
122 Variable<T, typelist::proj<I, A>>,
123 Variable<T, typelist::proj<I, B>>>;
124
125 public:
126 using type =
127 intlist::map_t<f, std::make_index_sequence<typelist::length<Dst>>>;
128};

131/// @tparam Op operator. See available operators in vm/operators.hpp

144 public:
145 using type =

180/// @brief Generate macro for dot product of two vectors.
181/// @tparam Dst destination register identifier.
182template <typename MAJOR_TYPE,
183 CharSequence Dst,
184 typelist::List A,
185 typelist::List B>
186 requires(typelist::length<A> == typelist::length<B>)
187class make_dot
188{
189 using dst_var = Variable<MAJOR_TYPE, Dst>;
190
191 template <std::uint8_t I>
192 using f = Compute<ops::fma,
193 dst_var,
194 Variable<MAJOR_TYPE, typelist::at<I, A>>,
195 Variable<MAJOR_TYPE, typelist::at<I, B>>,
196 dst_var>;
197
198 public:
199 using type = typelist::cons<
200 TMOV_CS(Dst, CS(0)),
201 intlist::map_t<f, std::make_index_sequence<typelist::length<A>>>>;
202};

204/// @brief Generate macro for cross product of two vectors[3].
205template <typename MAJOR_TYPE,
206 typelist::List Dst,
207 typelist::List A,
208 typelist::List B,
209 CharSequence TMP>
210 requires(typelist::length<A> == 3 && typelist::length<B> == 3 &&
211 typelist::length<Dst> == 3)
212class make_cross
213{
214 template <typename R, std::uint8_t I>
215 using tvar_idx = Variable<MAJOR_TYPE, typelist::at<I, R>>;
216
217 template <std::uint8_t K, std::uint8_t I, std::uint8_t J>
218 using s = type_sequence<Compute<std::multiplies<MAJOR_TYPE>,
219 Variable<MAJOR_TYPE, TMP>,
220 tvar_idx<A, J>,
221 tvar_idx<B, I>>,
222 Compute<ops::fms,
223 tvar_idx<Dst, K>,
224 tvar_idx<A, I>,
225 tvar_idx<B, J>,
226 Variable<MAJOR_TYPE, TMP>>>;
227
228 public:
229 using type =
230 typelist::concat<type_sequence<s<0, 1, 2>, s<1, 2, 0>, s<2, 0, 1>>>;
231};

233/// @brief Matrix-vector product.
234///
235/// @tparam Mat list of lists.
236template <typename MAJOR_TYPE,
237 typelist::List Dst,
238 typelist::List Mat,
239 typelist::List Vec>
240 requires(typelist::length<Dst> == typelist::length<Vec> &&
241 typelist::length<Mat> == typelist::length<Vec>)
242class make_mat_vec
243{
244 template <typename R, std::uint8_t I>
245 using tvar_idx = Variable<MAJOR_TYPE, typelist::at<I, R>>;
246
247 template <typename R, std::uint8_t I, std::uint8_t J>
248 using tvar_ij = Variable<MAJOR_TYPE, typelist::at<J, typelist::at<I, R>>>;
249
250 template <std::uint8_t I, std::uint8_t J>
251 using f = Compute<ops::fma,
252 tvar_idx<Dst, I>,
253 tvar_ij<Mat, I, J>,
254 tvar_idx<Vec, J>,
255 tvar_idx<Dst, I>>;
256
257 template <std::uint8_t I>
258 using init_dst = Compute<std::identity, tvar_idx<Dst, I>, TVAR(0)>;
259
260 using idxs = std::make_index_sequence<typelist::length<Vec>>;
261
262 template <std::uint8_t I>
263 using make_ij = type_pair<std::integral_constant<std::uint8_t, I>, idxs>;
264
265 template <std::uint8_t I>
266 class fi
267 {
268 template <std::uint8_t J>
269 using fij = f<I, J>;
270
271 public:
272 using type = intlist::map_t<fij, idxs>;
273 };
274
275 template <std::uint8_t I>
276 using fi_t = fi<I>::type;
277
278 public:
279 using type = typelist::join<intlist::map_t<init_dst, idxs>,
280 typelist::concat<intlist::map_t<fi_t, idxs>>>;
281};
