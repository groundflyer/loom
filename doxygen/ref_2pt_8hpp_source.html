66template <unsigned idx, std::floating_point T, typename Prim, typename... Rest>
67std::pair<unsigned, Intersection<T>>
68hider_impl(const std::pair<unsigned, Intersection<T>>& prev,
69 const Ray<T>& ray,
70 const Prim& prim,
71 const Rest&... rest) noexcept
72{
73 const Intersection<T> prev_isect = prev.second;
74 const Intersection<T> current_isect = intersect(ray, prim);
75 const std::pair<unsigned, Intersection<T>> current{idx, current_isect};
76 const std::pair<unsigned, Intersection<T>>& closest =
77 prev_isect ? (prev_isect.t < current_isect.t ? prev : current) :
78 current;
79
80 if constexpr(sizeof...(Rest) > 0)
81 return hider_impl<idx + 1>(closest, ray, rest...);
82 else
83 return closest;
84}
85} // namespace detail

118template <std::floating_point T>
119static Vec3<T>
120sample_hemisphere(const Vec3<T>& n, const Vec2<T>& u) noexcept
121{
122 const auto& [u1, u2] = u;
123 const T r1 = static_cast<T>(2) * Pi<T> * (u1 - T{0.5});
124 const T r2s = std::sqrt(u2);
125 const auto [x, y] = basis(n);
126 return normalize(x * std::cos(r1) * r2s + y * std::sin(r1) * r2s +
127 n * std::sqrt(1 - u2));
128}

130/// @brief Pixel position to film sample space.
131template <std::floating_point T>
132struct ToSampleSpace
133{
134 Vec2<T> res;
135 T ratio;
136
137 public:
138 ToSampleSpace(const Vec2u& _res) noexcept :
139 res(_res), ratio(res[0] / res[1])
140 {
141 }
142
143 Vec2<T>
144 operator()(const Vec2u& pos) const noexcept
145 {
146 const auto& [x, y] = pos;
147 const auto [w, h] = res;
148 const T sx = ratio * (static_cast<T>(x) / w - static_cast<T>(0.5));
149 const T sy = static_cast<T>(y) / h - static_cast<T>(0.5);
150 return Vec2<T>{sx, sy};
151 }

154/// @brief Fresnel term approximation.
155/// @param cd cosine theta.
156/// @param eta relative index of refraction.
157///
158/// See @cite 10.1111:1467-8659.1320121
159template <std::floating_point T>
160static T
161schlick(const T cd, const T eta) noexcept
162{
163 T r0 = (1 - eta) / (1 + eta);
164 r0 *= r0;
165 return r0 + (1 - r0) * static_pow<5>(1 - cd);
166}

377/// @brief Distribution based on Trowbridge-Reitz @cite Trowbridge:75.
378/// @param dotWmWg dot product of normal and microfacet.
379/// @param alpha surface roughness.
380///
381/// See @cite Trowbridge:75 and @cite 10.2312:EGWR/EGSR07/195-206
382/// Slightly modified for beautification.
383static float
384gtr(const float dotWmWg, const float alpha) noexcept
385{
386 return alpha / (alpha + 1 / (dotWmWg * dotWmWg) - 1);
387}

389/// @brief Microfacet geometry attenuation factor.
390/// @param cos_theta cosine of angle.
391/// @param alpha surface rougness.
392///
393/// See @cite 10.2312:EGWR/EGSR07/195-206
394static float
395gaf(const float cos_theta, const float alpha) noexcept
396{
397 return 2 / (1 + std::sqrt(loom::fma(alpha * alpha,
398 (1 / (cos_theta * cos_theta) - 1),
399 1.f)));
400}

402/// @brief Microfacet slope generation.
403///
404/// See @cite HeitzdEon2014
405static Vec2f
406get_slope(const float cos_theta, const Vec2f u) noexcept
407{
408 auto [sx, sy] = u;
409
410 if(cos_theta > 0.9999f)
411 {
412 const float r = std::sqrt(sx / (1 - sx));
413 const float phi = Pi<float> * 2 * (1 - sy);
414 return Vec2f{r * std::cos(phi), r * std::sin(phi)};
415 }
416
417 const float cos_theta2 = cos_theta * cos_theta;
418 const float B2 = (1 - cos_theta2) / std::max(cos_theta2, 1e-6f);
419 const float B = std::sqrt(B2);
420 const float a = 1 / B;
421 const float G1 = 2 / (1 + std::sqrt(1 + 1 / (a * a)));
422 const float A = 2 * sx / G1 - 1;
423 const float A2 = A * A;
424 const float denom = A2 - 1;
425 const float tmp =
426 1 / ((std::abs(denom) < 1e-6f) ? std::copysign(1e-6f, denom) : denom);
427 const float D = std::sqrt(B2 * tmp * tmp - (A2 - B2) * tmp);
428 const float Btmp = B * tmp;
429 const float slopex1 = Btmp - D;
430 const float slopex2 = Btmp + D;
431
432 const float retx = (A < 0 || slopex2 > 1 / B) ? slopex1 : slopex2;
433
434 float S;
435 if(sy > .5f)
436 {
437 S = 1;
438 sy = 2 * (sy - .5f);
439 }
440 else
441 {
442 S = -1;
443 sy = 2 * (.5f - sy);
444 }
445
446 const float z =
447 sy * loom::fma(loom::fma(sy, .27385f, -.73369f), sy, .46341f) /
448 loom::fma(loom::fma(loom::fma(sy, .093073f, .309420f), sy, -1.f),
449 sy,
450 .597999f);
451 const float rety = S * z * std::sqrt(1 + retx * retx);
452
453 return Vec2f{retx, rety};
454}

456/// @brief Microfacet generation.
457/// @param wi incident direction.
458/// @param alpha surface roughness.
459///
460/// See @cite HeitzdEon2014
461static Vec3f
462microfacet(Vec3f wi, const Vec2f& alpha, const Vec2f u) noexcept
463{
464 const auto [alphax, alphay] = alpha;
465
466 wi[0] *= alphax;
467 wi[1] *= alphay;
468
469 wi = normalize(wi);
470
471 const auto [wix, wiy, wiz] = wi;
472
473 const auto [sx, sy] = get_slope(wiz, u);
474
475 const float lensq = std::sqrt(wix * wix + wiy * wiy);
476 const bool tb = lensq > 1e-12f;
477 const float cosphi = tb ? wix / lensq : 1.f;
478 const float sinphi = tb ? wiy / lensq : 0.f;
479
480 const float nx = loom::fma(sinphi, sy, cosphi * sx) * alphax;
481 const float ny = loom::fma(cosphi, sy, -sinphi * sx) * alphay;
482
483 return make_normal(-nx, -ny, 1.f);
484}

486/// @brief incident projected roughness
487static float
488anisorough_i(const Vec3f& wi,
489 const Vec3f& tu,
490 const Vec3f& tv,
491 const Vec2f alpha) noexcept
492{
493 const float cs = dot(wi, tu);
494 const float sn = dot(wi, tv);
495
496 const auto [alphax, alphay] = alpha;
497
498 return std::sqrt(cs * cs * alphax * alphax + sn * sn * alphay * alphay);
499}
500
501/// @brief projected on surface rougness.
502static float
503anisorough_n(const Vec3f& wm,
504 const Vec3f& wg,
505 const Vec3f& tu,
506 const Vec3f& tv,
507 const float dotWmWg,
508 const Vec2f& alpha) noexcept
509{
510 const Vec3f ph = normalize(wm - wg * dotWmWg);
511 const float cu = dot(ph, tu);
512 const float cv = dot(ph, tv);
513
514 const auto [alphax, alphay] = alpha;
515
516 return 1 / (cu * cu / alphax + cv * cv / alphay);
517}

539 static constexpr unsigned ctr_incr = 39916801;

550 static ret_t
551 radiance(const unsigned max_depth,
552 const Ray<float>& ray,
553 const Counter& c,
554 const unsigned depth = 0,
555 unsigned ray_count = 0) noexcept
556 {
557 if(depth >= max_depth) return {{}, ray_count};
558
559 const auto [prim_id, isect] = std::apply(Hider{ray}, Scene::geometry);
560 ray_count++;
561
562 if(!isect) return {{}, ray_count};
563
564 const Vec3f p = ray.origin + ray.dir * isect.t;
565 const auto [n, t] = visit_at(
566 prim_id,
567 [&p](const auto& prim) noexcept
568 {
569 return std::pair{normal(p, prim), tangent(p, prim)}
570 ;
571 },
572 Scene::geometry,
573 constant(std::pair{Vec3f{0, 1, 0}, Vec3f{0, 1, 0}}));
574
575 return visit_at(Gallery::mat_idxs[prim_id],
576 Shading{
577 max_depth,
578 c,
579 depth + 1,
580 ray_count,
581 p,
582 n,
583 t,
584 ray.dir
585 },
586 Gallery::materials,
587 constant(ret_t{RGB{1, 0, 1}, ray_count}));
588 }

590 /// @brief Direct lighting estimator.
591 /// @return light PDF, cos(theta) and direction to light.
592 static std::pair<std::optional<std::tuple<float, float, Vec3f>>, unsigned>
593 direct(const Vec3f& p, const Vec3f& n, const Counter& c) noexcept
594 {
595 const Vec3f lp =
596 sample(sample_u(threefry2x32r32<key[0], key[1]>(c)), Scene::light);
597 const Vec3f d = normalize(lp - p);
598 const float cos_theta = dot(n, d);
599
600 if(cos_theta < 0) return {{}, 0}; // haven't trace
601
602 const auto [prim_id, isect] =
603 std::apply(Hider{bias_ray(p, d)}, Scene::geometry);
604
605 if(!isect || prim_id != 11) return {{}, 1}; // traced one ray
606
607 const float l = length2(Scene::light.position - p);
608 constexpr float pa =
609 Scene::light.radius * Scene::light.radius * Pi<float>;
610
611 return {
612 {{l / (std::abs(dot(Scene::light.normal, d)) * pa), cos_theta, d}},
613 1u};
614 }

616 ret_t
617 lambert(const RGB<>& color) const noexcept
618 {
619 const Vec3f d =
620 sample_hemisphere(n, sample_u(threefry2x32r32<key[0], key[1]>(c)));
621
622 auto cc = bump_ctr(c, ctr_incr);
623
624 auto [L, rc] =
625 radiance(max_depth, bias_ray(p, d), cc, depth, ray_count);
626
627 cc = bump_ctr(cc, ctr_incr);
628
629 const auto [r, _] = threefry2x32r32<key[0], key[1]>(cc);
630
631 cc = bump_ctr(cc, ctr_incr);
632
633 if(gen_canonical(r) < p_nee)
634 {
635 const auto [dr, drc] = direct(p, n, cc);
636 rc += drc;
637 if(dr)
638 {
639 const auto [p_l, ctl, _] = dr.value();
640 L += Scene::Gallery::Emission::value * ctl / (p_l * p_nee);
641 }
642 }
643
644 return {color * L, rc};
645 }

673 ret_t
674 operator()(const Gallery::Glass& /* mat */) const noexcept
675 {
676 const float cos_theta = dot(n, -i);
677 const bool into = cos_theta > 0;
678 const Vec3f nfn = into ? n : -n;
679 const float eta = compute_eta(Gallery::Glass::ior, into);
680 const float f = reflectance(cos_theta, eta);
681 const float ddn = dot(i, nfn);
682 const float cos2t = 1 - eta * eta * (1 - ddn * ddn);
683
684 auto refraction = [&]() noexcept -> ret_t
685 {
686 const auto [L, rc] =
687 radiance(max_depth,
688 bias_ray(p, refract(nfn, i, eta, ddn, cos2t)),
689 bump_ctr(c, ctr_incr),
690 depth);
691 return {L / (eta * eta), rc};
692 };
693
694 if(cos2t < 0) // total internal reflection
695 return reflection();
696
697 const auto [sx, _] = threefry2x32r32<key[0], key[1]>(c);
698
699 if(gen_canonical(sx) < f)
700 return reflection();
701 else
702 return refraction();
703 }

722 ret_t
723 operator()(const Gallery::Metal&) const noexcept
724 {
725 const Vec3f wi = -i;
726 const Vec2f alpha = Gallery::Metal::alpha;
727 const Vec3f b = cross(n, t);
728
729 const auto [wmx, wmy, wmz] =
730 microfacet(wi, alpha, sample_u(threefry2x32r32<key[0], key[1]>(c)));
731 const Vec3f wm = normalize(t * wmx + b * wmy + n * wmz);
732 const Vec3f wo = reflect(wm, i, dot(wm, wi));
733
734 const float dotWgWm = dot(n, wm);
735 const float dotWgWo = dot(n, wo);
736
737 if(dotWgWo < 0) return {RGB{0}, ray_count};
738
739 const float alpha_n = anisorough_n(wm, n, t, b, dotWgWm, alpha);

741

756 const Vec3f wm = normalize(wi + l);
757 const float dotWgWm = dot(n, wm);
758 const float alpha_n = anisorough_n(wm, n, t, b, dotWgWm, alpha);

760 const float f = gtr(dotWgWm, alpha_n);
761 const float dotWgWi = dot(n, wi);
762 L += Scene::Gallery::Emission::value * f *

764 (4 * dotWgWi * p_nee * p_l);
765 }
766 }
767
768 return {Gallery::Metal::color * L, rc};
769 }

783 return reflection();
784 else

792
793 const Camera cam = Scene::camera();
794
795 const unsigned m_width;
796 const ToSampleSpace<float> m_sample_space;
797 const float m_pw; // pixel width
798 const float m_threshold;
799 const unsigned m_max_depth;
800
801 public:
802 RenderPixel(const Vec2u& res,
803 const float threshold,
804 const unsigned max_depth,
805 const float blur) noexcept :
806 m_width(res[0]), m_sample_space(res),
807 m_pw(blur * m_sample_space.ratio / m_sample_space.res[0]),
808 m_threshold(threshold), m_max_depth(max_depth)
809 {
810 }
811
812 template <typename Pixel>
813 std::pair<std::size_t, std::size_t>
814 operator()(Pixel val) const noexcept
815 {
816 auto [pos, pix] = val;
817 const auto [px, py] = pos;
818 const Vec2f pp = m_sample_space(pos); // pixel position
819 const std::uint32_t c0 = px * m_width + py;
820 unsigned ns = 0;
821 unsigned prc = 0; // pixel ray count
822 float mean = .0f;
823 float m2 = .0f;
824
825 RGB r{};
826
827 while(ns < min_samples ||
828 std::sqrt(m2 / static_cast<float>(ns - 1)) / ns >= m_threshold)
829 {
830 const auto [v, rc] = Shading::radiance(
831 m_max_depth,
832 cam(jitter_sample_pos(m_pw,
833 pp,
834 sample_u(threefry2x32r32<key[0], key[1]>(
835 Counter{c0, ns})))),
836 Counter{c0, ns + min_samples + 1001});
837 const float x = std::min(luminance(v), 1.f);
838 ++ns;
839 const float delta = x - mean;
840 mean += delta / ns;
841 const float delta2 = x - mean;
842 m2 += delta * delta2;
843
844 r += v;
845 prc += rc;
846 }
847
848 pix = RGBA{r / static_cast<float>(ns), 1};
849
850 return {1, prc};
851 }
