43/// @brief Wraps char[@p N].
44template <std::size_t _N>
45struct StringLiteral
46{
47 // cut off null terminator
48 static constexpr std::size_t N = _N - 1;
49
50 constexpr StringLiteral(const char (&str)[_N])
51 {
52 for(std::size_t i = 0; i < N; ++i) value[i] = str[i];
53 }
54
55 char value[N]{};
56
57 constexpr char
58 operator[](const std::size_t i) const
59 {
60 return value[i];
61 }
62
63 constexpr std::size_t
64 size() const
65 {
66 return N;
67 }
68
69 constexpr std::string_view
70 view() const
71 {
72 return std::string_view(value, N);
73 }
74};

183/// @brief Converts character into the corresponding integer number.
184///
185/// @param c character to convert. Valid set is [0-9,a-f,A-F] for base 16, [0-9]
186/// for 10 and (0, 1) for 2.
187/// @param the number base. Valid values are 16, 10, 2.
188template <std::integral Int = int>
189constexpr std::optional<Int>
190c2i(char c, Int base = 10) noexcept
191{
192 if(base == 16)
193 {
194 if(is_abcdef_lower(c)) return static_cast<Int>(c - 87);
195 if(is_abcdef_upper(c)) return static_cast<Int>(c - 55);
196 }
197
198 if(is_digit(c))
199 {
200 const Int num = static_cast<Int>(c - 48);
201 if(num < base) return num;
202 }
203
204 return {};
205}

207template <typename T>
208 requires(std::signed_integral<T> || std::floating_point<T>)
209constexpr std::pair<T, std::string_view>
210parse_sign(std::string_view s) noexcept
211{
212 T sign = static_cast<T>(1);
213
214 if(s.starts_with('-') || s.starts_with('+'))
215 {
216 if(s.starts_with('-')) sign *= static_cast<T>(-1);
217 s = s.substr(1);
218 }
219
220 return {sign, s};
221}
222
223template <std::integral Int>
224constexpr std::optional<Int>
225parse_num(std::string_view s, Int base) noexcept
226{
227 if(s.empty()) return {};
228
229 Int ret = 0;
230 Int exp = 0;
231 for(auto it = s.crbegin(); it != s.crend(); ++it)
232 {
233 const auto o = c2i<Int>(*it, base);
234 if(!o) return {};
235
236 ret += o.value() * smath::pow(base, exp);
237 ++exp;
238 }
239
240 return ret;
241}
242
243template <std::signed_integral Int>
244constexpr std::optional<Int>
245parse_signed(std::string_view s, Int base) noexcept
246{
247 auto [sign, rest] = parse_sign<Int>(s);
248
249 if(rest.empty()) return {};
250
251 std::optional<Int> num = parse_num<Int>(rest, base);
252 if(num) return num.value() * sign;
253
254 return {};
255}

257constexpr std::pair<unsigned, std::string_view>
258detect_base(std::string_view s) noexcept
259{
260 if(any_string(s.substr(0, 2), "0x", "0X"))
261 return {16, s.substr(2)};
262 else if(any_string(s.substr(0, 2), "0b", "0B"))
263 return {2, s.substr(2)};
264
265 return {10, s};
266}

284/// @brief For strings that could have prefixes like "0x" or "0b".
285template <std::integral Int = int>
286struct s2i_base_prefix_fn
287{
288 constexpr std::optional<Int>
289 operator()(std::string_view s) const noexcept
290 {
291 if(s.empty()) return {};
292
293 if constexpr(std::signed_integral<Int>)
294 {
295 auto [sign, rest] = parse_sign<Int>(s);
296 const auto [base, rest2] = detect_base(rest);
297 const std::optional<Int> num = parse_num<Int>(rest2, base);
298 if(num) return num.value() * sign;
299 }
300 else
301 {
302 const auto [base, rest] = detect_base(s);
303 return parse_num<Int>(rest, base);
304 }
305
306 return {};
307 }

312/// @brief Inaccurate constexpr analogue of std::stof.
313template <std::floating_point T = float>
314struct s2f_fn
315{
316 constexpr std::optional<T>
317 operator()(std::string_view word) const noexcept
318 {
319 auto [sign, s] = parse_sign<T>(word);
320
321 if(s.empty()) return {};
322
323 const std::size_t sep_pos = s.find('.');
324 const std::size_t ep1 = s.find('e');
325 const std::size_t ep2 = s.find('E');
326 const std::size_t exp_pos = std::min(ep1, ep2);
327
328 const bool has_sep = sep_pos != std::string_view::npos;
329 const bool has_exp = exp_pos != std::string_view::npos;
330
331 std::string_view int_part = has_sep ?
332 s.substr(0, sep_pos) :
333 (has_exp ? s.substr(0, exp_pos) : s);
334
335 std::string_view frac_part =
336 has_sep ? (has_exp ? s.substr(sep_pos + 1, exp_pos - sep_pos - 1) :
337 s.substr(sep_pos + 1)) :
338 "";
339
340 std::string_view exp_part = has_exp ? s.substr(exp_pos + 1) : "";
341
342 T ret{};
343
344 if(!int_part.empty())
345 {
346 const auto oi = parse_num<long>(int_part, 10);
347 if(!oi.has_value()) return {};
348
349 ret = static_cast<T>(oi.value());
350 }
351
352 if(!frac_part.empty())
353 {
354 const auto oi = parse_num<long>(frac_part, 10);
355 if(!oi.has_value()) return {};
356
357 const auto frac =
358 oi.value() * smath::pow(static_cast<T>(10),
359 -static_cast<int>(frac_part.length()));
360
361 ret += frac;
362 }
363
364 if(!exp_part.empty())
365 {
366 const auto oi = parse_signed<long>(exp_part, 10);
367 if(!oi.has_value()) return {};
368
369 ret *= smath::pow(static_cast<T>(10), oi.value());
370 }
371
372 return ret * sign;
373 }
374
375 COMPOSE_OP
376};

424rtrim(std::string_view s) noexcept
425{
426 auto it = std::find_if(s.rbegin(),
427 s.rend(),
