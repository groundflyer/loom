62struct ComposeOp
63{
64 template <typename LHS, typename RHS>
65 struct Invoker
66 {
67 LHS lhs;
68 RHS rhs;
69
70 constexpr Invoker() noexcept {}
71 constexpr Invoker(const Invoker& other) noexcept :
72 lhs(other.lhs), rhs(other.rhs)
73 {
74 }
75 constexpr Invoker(Invoker&& other) noexcept :
76 lhs(std::move(other.lhs)), rhs(std::move(other.rhs))
77 {
78 }
79 constexpr Invoker(LHS _lhs, RHS _rhs) noexcept : lhs(_lhs), rhs(_rhs) {}
80
81 constexpr Invoker&
82 operator=(const Invoker& other) noexcept
83 {
84 lhs = other.lhs;
85 rhs = other.rhs;
86 return *this;
87 }
88
89 constexpr Invoker&
90 operator=(Invoker&& other) noexcept
91 {
92 lhs = std::move(other.lhs);
93 rhs = std::move(other.rhs);
94 return *this;
95 }
96
97 template <typename... Args>
98 constexpr decltype(auto)
99 operator()(Args&&... args) const
100 {
101 return lhs(rhs(std::forward<Args>(args)...));
102 }
103
104 template <typename... Args>
105 constexpr decltype(auto)
106 operator()(Args&&... args)
107 {
108 return lhs(rhs(std::forward<Args>(args)...));
109 }
110 };
111
112 template <typename LHS, typename RHS>
113 constexpr decltype(auto)
114 operator()(LHS lhs, RHS rhs) const
115 {
116 return Invoker(lhs, rhs);
117 }
118};

120template <typename BinaryOp, typename LeftOperand>
121struct fold_wrapper
122{
123 BinaryOp op;
124 LeftOperand operand;
125
126 constexpr fold_wrapper(BinaryOp _op, const LeftOperand& _operand) :
127 op(_op), operand(_operand)
128 {
129 }
130
131 constexpr fold_wrapper() = delete;
132
133 constexpr fold_wrapper(const fold_wrapper&) = delete;
134
135 template <typename RightOperand>
136 constexpr decltype(auto)
137 operator%(const fold_wrapper<BinaryOp, RightOperand>& rhs) const noexcept
138 {
139 return fold_wrapper<
140 BinaryOp,
141 std::invoke_result_t<BinaryOp, LeftOperand, RightOperand>>(
142 op,
143 op(operand, rhs.operand));
144 }
145};
146} // namespace detail

148/// @brief Accumulate @p args with @p op.
149template <typename BinaryOp, typename... Args>
150constexpr decltype(auto)
151fold(BinaryOp&& op, Args&&... args) noexcept
152{
153 return (... % detail::fold_wrapper(op, args)).operand;

156/// @brief Compose functions, e.g. compose(a, b, c) = a(b(c())).
157template <typename... Fs>
158constexpr decltype(auto)
159compose(Fs&&... fs)
160{
161 return fold(detail::ComposeOp{}, std::forward<Fs>(fs)...);

259/// @brief Create a functional object that takes its two arguments in the
260/// reverse order of @p f.
261template <typename BinaryF>
262constexpr decltype(auto)
263flip(BinaryF f) noexcept
264{
265 return [=](auto&& rhs, auto&& lhs) constexpr
266 { return std::invoke(f, lhs, rhs); };
267}

269/// @brief Get the @p Ith argument.
270template <std::size_t I, typename Head, typename... Tail>
271constexpr decltype(auto)
272select(Head&& head, Tail&&... tail) noexcept
273 requires(I <= sizeof...(tail))
274{
275 if constexpr(I == 0ul)
276 return head;
277 else
