41/// @brief FMA that fallbacks to expression @f$x * y + z@f$ if fast FMA

46{

114/// @brief Compute inversion number.
115///
116/// Complexity @f$O(N^{2})@f$.
117/// https://en.wikipedia.org/wiki/Inversion_(discrete_mathematics)
118template <typename Iterable>
119constexpr unsigned
120inversion_number(const Iterable& input) noexcept
121{
122 unsigned ret = 0;
123
124 for(auto i = input.cbegin(); i != input.cend() - 1; ++i)
125 for(auto j = std::next(i); j != input.cend(); ++j)
126 if(*i > *j) ++ret;
127
128 return ret;
129}

131/// @brief Test whether the input contains equal elements.
132///
133/// Complexity @f$O(N^{2})@f$.
134template <typename Iterable>
135constexpr bool
136has_equal(const Iterable& input) noexcept
137{
138 for(auto i = input.cbegin(); i != input.cend() - 1; ++i)
139 for(auto j = std::next(i); j != input.cend(); ++j)
140 if(*i == *j) return true;
141 return false;
142}

219/// @brief Rise a number to a constant power.
220///
221/// @tparam exp power exponent.
222template <unsigned exp, typename T>
223constexpr T
224static_pow(const T value) noexcept
225{
226 if(!exp) { return 1; }
227 else
228 {
229 const T base = static_pow<exp / 2>(value);
230 if constexpr(is_even(exp))
231 return base * base;
232 else
233 return base * base * value;
234 }
235}

246/// @brief Compare floating-point values for equality using epsilon.
247///
248/// The implementation is taken from
249/// https://en.cppreference.com/w/cpp/types/numeric_limits/epsilon
250///
251/// @param ulp spacing between adjascent numbers (see
252/// https://en.wikipedia.org/wiki/Unit_in_the_last_place). @anchor ulp
253template <std::floating_point T, std::integral ULP>
254constexpr bool
255almost_equal(const T a, const T b, const ULP ulp)
256{
257 // the machine epsilon has to be scaled to the magnitude of the values used
258 // and multiplied by the desired precision in ULPs (units in the last place)
259 // unless the result is subnormal
260 T amb{a - b};
261 T apb{a + b};
262 if(std::is_constant_evaluated())
263 {
264 amb = smath::abs(amb);
265 apb = smath::abs(apb);
266 }
267 else
268 {
269 amb = std::abs(amb);
270 apb = std::abs(apb);
271 }
272 return amb <= std::numeric_limits<T>::epsilon() * apb * ulp ||
273 amb < std::numeric_limits<T>::min();
274}

306/// @brief Solve quadratic equation @f$ax^{2} + bx + c@f$ for non-complex cases.
307/// @return pair of bool and the smallest of two positive roots.
308template <std::floating_point T>
309constexpr std::pair<bool, T>
310quadratic(const T a, const T b, const T c) noexcept
311{
312 if(almost_equal(a, T{0}, 10))
313 {
314 if(!almost_equal(b, T{0}, 10)) return {true, -c / b};
315 return {false, -1};
316 }
317 const T D = b * b - T{4} * a * c;
318
319 if(D < 0) return {false, -1};
320
321 T sqrtD{};
322 if(std::is_constant_evaluated())
323 sqrtD = smath::sqrt(D);
324 else
325 sqrtD = std::sqrt(D);
326
327 const T temp = T{-0.5} * (b < 0 ? b - sqrtD : b + sqrtD);
328
329 const T x1 = temp / a;
330 const T x2 = c / temp;
331
332 const T& x = (x1 < 0) ? x2 : ((x2 < 0) ? x1 : std::min(x1, x2));
333 return std::pair{((D * x) >= 0), x};
334}
