39static std::string
40get_cpu_model() noexcept
41{
42 std::uint32_t eax, ebx, ecx, edx;
43 __cpuid(1, eax, ebx, ecx, edx);
44
45 std::uint32_t stepping = eax & 0xf;
46 std::uint32_t model = (eax >> 4) & 0xf;
47 std::uint32_t family = (eax >> 8) & 0xf;
48 std::uint32_t ext_model = (eax >> 16) & 0xf;
49 std::uint32_t ext_family = (eax >> 20) & 0xff;
50
51 std::uint32_t display_family =
52 (family == 0xf) ? family + ext_family : family;
53 std::uint32_t display_model =
54 (family == 6 || family == 0xf) ? ((ext_model << 4) | model) : model;
55
56 __cpuid(0, eax, ebx, ecx, edx);
57
58 char vendor[13]{};
59
60 std::memcpy(vendor, &ebx, 4);
61 std::memcpy(vendor + 4, &edx, 4);
62 std::memcpy(vendor + 8, &ecx, 4);
63
64 return fmt::format("{} fam{} mod{} step{}",
65 rtrim(vendor),
66 display_family,
67 display_model,
68 stepping);
69}

71template <typename Dur>
72static std::string
73format_duration(Dur d,
74 const unsigned z,
75 const unsigned f,
76 const bool always_mins = false) noexcept
77{
78 const auto mins = std::chrono::duration_cast<std::chrono::minutes>(d);
79 d -= mins;
80
81 if(mins.count() || always_mins)
82 return std::format("{:02}:{:0{}.{}f}",
83 mins.count(),
84 std::chrono::duration<double>(d).count(),
85 z,
86 f);
87 else
88 return std::format("{:0{}.{}f}s",
89 std::chrono::duration<double>(d).count(),
90 z,
91 f);
92}

103template <typename Dur = std::chrono::steady_clock::duration, typename... Bms>
104static std::array<bm_result<Dur>, sizeof...(Bms)>
105run_benchmarks(Logger& logger,
106 std::FILE* file,
107 std::size_t num_runs,
108 Bms... bms)
109{
110 auto decorate = [&](auto&& bm)
111 {
112 const std::string i{bm.info()};
113 Dur avg{};
114 for(std::size_t j = 0; j < num_runs; ++j)
115 {
116 logger.info("Benchmarking {} [{}/{}]...", i, j + 1ul, num_runs);
117 const Dur dur = bm();
118 avg += dur;
119 logger.info("Result {}", format_duration(dur, 6, 3));
120 if(file)
121 fmt::print(
122 file,
123 "{:%FT%TZ},{},clang-{},fast-math:{},{},{},{},{:09.6f}\n",
124 std::chrono::floor<std::chrono::seconds>(
125 std::chrono::system_clock::now()),
126 REPO_HASH,
127 rtrim(__clang_version__),
128 fast_math,
129 get_cpu_model(),
130 vm::simd::info::value,
131 i,
132 std::chrono::duration<double>(dur).count());
133 }
134
135 return bm_result<Dur>{std::move(i), avg / num_runs};
136 };
137
138 return {decorate(bms)...};
139}

143template <typename Dur, std::size_t N>
144static void
145print_results(std::array<bm_result<Dur>, N> results) noexcept
146{
147 std::ranges::sort(results,
148 std::less<std::size_t>{},
149 [](const auto& e) noexcept { return e.second.count(); });
150
151 const Dur fastest_dur = results.front().second;
152 const Dur slowest_dur = results.back().second;
153
154 std::size_t id_width = 10;
155
156 for(const auto& [id, _]: results) id_width = std::max(id_width, id.size());
157
158 const bool has_mins =
159 std::chrono::duration_cast<std::chrono::minutes>(slowest_dur).count();
160 const unsigned time_width = has_mins ? 9 : 7;
161 const unsigned sd_width =
162 magnitude(static_cast<unsigned>(static_cast<double>(
163 slowest_dur.count() / static_cast<double>(fastest_dur.count())))) +
164 8;
165
166 fmt::print("| {:<{}} | {:^{}} | {:>{}} |\n",
167 "Benchmark",
168 id_width,
169 "Time",
170 time_width,
171 "Slowdown",
172 sd_width);
173
174 fmt::print("|:{0:-^{1}}|:{0:-^{2}}:|:{0:-^{3}}:|\n",
175 "",
176 id_width + 1,
177 time_width,
178 sd_width);
179
180 for(const auto& [id, dur]: results)
181 fmt::print("| {:<{}} | {} | {:>{}.2f}x |\n",
182 id,
183 id_width,
184 format_duration(dur, 6, 3, has_mins),
185 static_cast<double>(dur.count()) /
186 static_cast<double>(fastest_dur.count()),
187 sd_width - 1);
188}
189} // namespace fn
